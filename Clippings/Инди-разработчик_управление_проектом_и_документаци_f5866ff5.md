Единый источник правды о состоянии проекта, который:

```bash
#!/bin/bash
# scripts/update-progress.sh

DATE=$(date +%Y-%m-%d)
PROGRESS_FILE="docs/01-ПРОГРЕСС.md"

# Создаем архив предыдущей версии
cp "$PROGRESS_FILE" "docs/archives/progress-$DATE.md"

# Обновляем дату в файле
sed -i "s|**Последнее обновление:**.*|**Последнее обновление:** $DATE|" "$PROGRESS_FILE"

echo "Прогресс обновлен для $DATE"
```

## 🛠 **СКРИПТЫ ДЛЯ РАБОТЫ С ДОКУМЕНТАЦИЕЙ**

### **1. Создаем скрипт обновления прогресса:** `scripts/docs-update.sh`

```bash
#!/bin/bash

# scripts/docs-update.sh
# Обновляет файл прогресса и создает ежедневную заметку

DOCS_DIR="docs"
CURRENT_PROGRESS="$DOCS_DIR/01-ТЕКУЩИЙ-ПРОГРЕСС.md"
DAILY_DIR="$DOCS_DIR/daily"
TODAY=$(date +%Y-%m-%d)
DAILY_FILE="$DAILY_DIR/$TODAY.md"
ARCHIVE_DIR="$DOCS_DIR/archive/progress"

# Создаем директории если их нет
mkdir -p "$DAILY_DIR"
mkdir -p "$ARCHIVE_DIR"

# Архивируем предыдущую версию прогресса (если есть)
if [ -f "$CURRENT_PROGRESS" ]; then
    ARCHIVE_FILE="$ARCHIVE_DIR/01-ТЕКУЩИЙ-ПРОГРЕСС-$(date +%Y-%m-%d-%H%M).md"
    cp "$CURRENT_PROGRESS" "$ARCHIVE_FILE"
    echo "📁 Архив прогресса создан: $(basename $ARCHIVE_FILE)"
fi

# Создаем или обновляем ежедневную заметку
if [ ! -f "$DAILY_FILE" ]; then
    cat > "$DAILY_FILE" << EOF
# Ежедневная заметка: $TODAY

## 🎯 Цели на сегодня

## 📝 Что сделал

## 🧩 Проблемы и решения

## 💡 Идеи на будущее

## 🔮 План на завтра
EOF
    echo "📓 Создана ежедневная заметка: $TODAY.md"
fi

# Открываем файлы в редакторе (или создаем новый прогресс)
if [ ! -f "$CURRENT_PROGRESS" ]; then
    # Создаем новый файл прогресса
    cat > "$CURRENT_PROGRESS" << EOF
# UNIVERSAL CREATIVE HUB — ТЕКУЩИЙ ПРОГРЕСС
**Последнее обновление:** $TODAY  
**Версия проекта:** 0.3.1
```

```bash
#!/bin/bash

# scripts/docs-archive.sh
# Архивирует старые файлы и чистит архив

DOCS_DIR="docs"
ARCHIVE_DIR="$DOCS_DIR/archive"
DAYS_TO_KEEP=30

echo "🧹 Очистка архива документации"

# Создаем директории архива если их нет
mkdir -p "$ARCHIVE_DIR/progress"
mkdir -p "$ARCHIVE_DIR/kanban"
mkdir -p "$ARCHIVE_DIR/daily"

# 1. Архивируем старые версии прогресса (если они не в архиве)
if [ -f "$DOCS_DIR/01-ТЕКУЩИЙ-ПРОГРЕСС-*.md" ]; then
    mv "$DOCS_DIR/01-ТЕКУЩИЙ-ПРОГРЕСС-"*.md "$ARCHIVE_DIR/progress/" 2>/dev/null || true
fi

# 2. Архивируем старые версии канбана
if [ -f "$DOCS_DIR/02-КАНБАН-ЗАДАЧ-*.md" ]; then
    mv "$DOCS_DIR/02-КАНБАН-ЗАДАЧ-"*.md "$ARCHIVE_DIR/kanban/" 2>/dev/null || true
fi

# 3. Удаляем старые архивы (старше 30 дней)
echo "🗑️  Удаляю архивы старше $DAYS_TO_KEEP дней..."

find "$ARCHIVE_DIR" -name "*.md" -type f -mtime +$DAYS_TO_KEEP -exec rm -v {} \;

# 4. Статистика
PROGRESS_COUNT=$(find "$ARCHIVE_DIR/progress" -name "*.md" | wc -l)
KANBAN_COUNT=$(find "$ARCHIVE_DIR/kanban" -name "*.md" | wc -l)
DAILY_COUNT=$(find "$DOCS_DIR/daily" -name "*.md" | wc -l)

echo ""
echo "📊 Статистика архивации:"
echo "   Прогресс: $PROGRESS_COUNT файлов"
echo "   Канбан: $KANBAN_COUNT файлов"
echo "   Ежедневных заметок: $DAILY_COUNT файлов"
echo ""
echo "✅ Архивация завершена"
```
