    ```typescript
    // StrudelCoreService.ts
    export class StrudelCoreService {
      private static instance: StrudelCoreService;
      private isInitialized = false;

      private constructor() {}

      public static getInstance(): StrudelCoreService {
        if (!StrudelCoreService.instance) {
          StrudelCoreService.instance = new StrudelCoreService();
        }
        return StrudelCoreService.instance;
      }

      public async initialize(): Promise<void> {
        if (this.isInitialized) {
          return;
        }
        // 1. Динамическая загрузка скрипта Strudel через CDN
        await this.loadScript('https://unpkg.com/@strudel/web@latest');
        // 2. Проверка, что глобальный объект Strudel доступен
        // @ts-ignore - глобальный объект появится после загрузки скрипта
        if (window.strudel || window.Strudel) {
          console.log('Strudel loaded successfully');
          this.isInitialized = true;
        } else {
          throw new Error('Failed to load Strudel library');
        }
      }

      private loadScript(src: string): Promise<void> {
        return new Promise((resolve, reject) => {
          const script = document.createElement('script');
          script.src = src;
          script.onload = () => resolve();
          script.onerror = () => reject(new Error(`Script load error for ${src}`));
          document.head.appendChild(script);
        });
      }

      // Метод для получения доступа к глобальному API Strudel
      public getStrudelAPI(): any {
        // @ts-ignore
        return window.strudel || window.Strudel || {};
      }

      // Вспомогательный метод для быстрого теста
      public async testPattern(patternCode: string): Promise<void> {
        await this.initialize();
        const api = this.getStrudelAPI();
        // Здесь потребуется разобраться с API, см. RESEARCH_PLAN_ESM.md
        console.log('Strudel API available:', api);
        // Пока просто выведем код
        console.log('Test pattern:', patternCode);
      }
    }
    ```
