function calculateNodeSize(node) {
    // Базовый размер от типа
    const typeScale = {
        'Chunk': 0.8,
        'Task': 1.2,
        'Component': 1.5,
        'Technology': 2.0,
        'Person': 2.2,
        'Decision': 1.8
    }[node.group] || 1.0;
    
    // Модификатор от количества связей
    const connectionCount = graphData.links.filter(l => 
        l.source.id === node.id || l.target.id === node.id
    ).length;
    
    // Логарифмическая шкала для связи (чтобы не было гигантских узлов)
    const connectionModifier = 1 + Math.log2(connectionCount + 1) * 0.2;
    
    return typeScale * connectionModifier;
}

**НЕМЕДЛЕННАЯ ДИАГНОСТИКА ПРОИЗВОДИТЕЛЬНОСТИ:**

```javascript
// Внедряем FPS монитор в консоль браузера на странице визуализации
console.log('🔍 ЗАПУСК FPS МОНИТОРИНГА');

let lastTime = performance.now();
let frames = 0;
let fpsHistory = [];
let minFps = 60;
let maxFps = 0;
let freezeDetected = 0;

function measureFPS() {
    frames++;
    const now = performance.now();
    const delta = now - lastTime;
    
    if (delta >= 1000) {
        const currentFps = frames;
        fpsHistory.push(currentFps);
        if (fpsHistory.length > 60) fpsHistory.shift();
        
        // Анализ
        minFps = Math.min(minFps, currentFps);
        maxFps = Math.max(maxFps, currentFps);
        
        // Детектор фризов (падение ниже 30 FPS)
        if (currentFps < 30) {
            freezeDetected++;
            console.warn(`⚠️ ФРИЗ ОБНАРУЖЕН! FPS: ${currentFps} (${freezeDetected}-й)`);
        }
        
        // Статистика
        const avgFps = fpsHistory.reduce((a, b) => a + b, 0) / fpsHistory.length;
        console.log(
            `📊 FPS: ${currentFps} | ` +
            `Min: ${minFps} | ` +
            `Max: ${maxFps} | ` +
            `Avg: ${avgFps.toFixed(1)} | ` +
            `Freezes: ${freezeDetected}`
        );
        
        frames = 0;
        lastTime = now;
    }
    requestAnimationFrame(measureFPS);
}

// Запускаем мониторинг
measureFPS();

// Дополнительно: мониторинг памяти (если доступно)
if (performance.memory) {
    setInterval(() => {
        console.log('🧠 Память:', {

            total: Math.round(performance.memory.totalJSHeapSize / 1048576) + ' MB',
            limit: Math.round(performance.memory.jsHeapSizeLimit / 1048576) + ' MB'
        });
    }, 5000);
}
```

```javascript
function calculateNodeSize(node, graphData) {
    // Иерархическая шкала типов
    const typeScale = {
        'Chunk': 0.7,        // Самые маленькие - атомарные единицы
        'Task': 1.0,          // Базовый уровень
        'Component': 1.4,     // Компоненты системы
        'Technology': 1.8,    // Технологии
        'Person': 2.0,        // Люди (важные узлы)
        'Decision': 1.6,      // Решения
        'default': 1.0
    };
    
    // Определяем тип узла (пробуем разные поля)
    const nodeType = node.group || node.type || node.label_type || 'default';
    const baseSize = typeScale[nodeType] || typeScale.default;
    
    // Подсчет связей для узла
    const connectionCount = graphData.links.filter(l => 
        (typeof l.source === 'object' ? l.source.id === node.id : l.source === node.id) ||
        (typeof l.target === 'object' ? l.target.id === node.id : l.target === node.id)
    ).length;
    
    // Логарифмический модификатор связей (чтобы не было гигантских узлов)
    const connectionModifier = 1 + Math.log2(connectionCount + 1) * 0.15;
    
    // Финальный размер
    return baseSize * connectionModifier;
}
```

```python
    async def get_graph_chunked(self, limit: int = 8000, chunk_size: int = 1000):
        """
        Получение графа чанками для оптимизации памяти
        """
        async with self.driver.session() as session:
            # Сначала получаем все узлы
            nodes_result = await session.run(
                "MATCH (n) RETURN n, labels(n)[0] as type LIMIT $limit",
                limit=limit
            )
            
            nodes = []
            async for record in nodes_result:
                node = record["n"]
                nodes.append({
                    "id": node.id,
                    "label": node.get("name", node.get("title", f"Node_{node.id}")),
                    "group": record["type"],
                    "type": record["type"]
                })
            
            # Получаем связи для этих узлов
            node_ids = [n["id"] for n in nodes]
            links_result = await session.run(
                """
                MATCH (a)-[r]->(b)
                WHERE id(a) IN $node_ids AND id(b) IN $node_ids
                RETURN id(a) as source, id(b) as target, type(r) as type
                """,
                node_ids=node_ids
            )
            
            links = []
            async for record in links_result:
                links.append({
                    "source": record["source"],
                    "target": record["target"],
                    "type": record["type"]
                })
            
            # Отправляем чанками
            def generate_chunks():
                yield json.dumps({"total_nodes": len(nodes), "total_links": len(links)}).encode() + b"\n"
                
                for i in range(0, len(nodes), chunk_size):
                    chunk = nodes[i:i+chunk_size]
                    yield json.dumps({"type": "nodes_chunk", "data": chunk}).encode() + b"\n"
                
                for i in range(0, len(links), chunk_size):
                    chunk = links[i:i+chunk_size]
                    yield json.dumps({"type": "links_chunk", "data": chunk}).encode() + b"\n"
                
                yield json.dumps({"type": "complete"}).encode() + b"\n"
            
            return StreamingResponse(
                generate_chunks(),
                media_type="application/x-ndjson"
            )
    
    async def get_graph_compressed(self, limit: int = 8000):
        """
        Получение сжатого графа
        """
        async with self.driver.session() as session:
            # Получаем данные
            result = await session.run(
                """
                MATCH (n)
                OPTIONAL MATCH (n)-[r]-(m)
                WITH n, collect(DISTINCT {
                    type: type(r),
                    target: id(m)
                }) as connections
                RETURN n, labels(n)[0] as type, connections
                LIMIT $limit
                """,
                limit=limit
            )
            
            nodes = []
            links = []
            
            async for record in result:
                node = record["n"]
                node_id = node.id
                nodes.append({
                    "id": node_id,
                    "label": node.get("name", node.get("title", f"Node_{node_id}")),
                    "group": record["type"]
                })
                
                for conn in record["connections"]:
                    if conn["target"]:
                        links.append({
                            "source": node_id,
                            "target": conn["target"],
                            "type": conn["type"]
                        })
            
            # Сжимаем
            json_str = json.dumps({"nodes": nodes, "links": links})
            compressed = gzip.compress(json_str.encode())
            
            return Response(
                content=compressed,
                media_type="application/json",
                headers={"Content-Encoding": "gzip"}
            )
```

```python
# semantic_clustering.py
import numpy as np
from sklearn.cluster import HDBSCAN
from sklearn.metrics.pairwise import cosine_similarity
from typing import List, Dict, Tuple
import asyncio
from dataclasses import dataclass

@dataclass
class ChunkCluster:
    id: str
    name: str
    chunks: List[str]  # IDs чанков
    centroid: np.ndarray
    theme: str
    keywords: List[str]
    size: int
    representative_chunk: dict

class HierarchicalChunkAggregator:
    """
    Аггрегация миллионов чанков в иерархическую структуру
    """
    
    def __init__(self, embedding_dim=1536):
        self.embedding_dim = embedding_dim
        self.clusters = []
        
    async def aggregate_chunks(self, chunks: List[dict], 
                               target_clusters: int = 1000) -> List[ChunkCluster]:
        """
        Аггрегация миллионов чанков в тысячи кластеров
        """
        print(f"🔄 Аггрегация {len(chunks)} чанков в {target_clusters} кластеров")
        
        # 1. Получаем эмбеддинги для всех чанков
        embeddings = np.array([chunk['embedding'] for chunk in chunks])
        
        # 2. Используем HDBSCAN для иерархической кластеризации
        clusterer = HDBSCAN(
            min_cluster_size=5,
            min_samples=3,
            metric='euclidean',
            cluster_selection_epsilon=0.5
        )
        
        # 3. Выполняем кластеризацию батчами для миллионов чанков
        batch_size = 10000
        all_labels = []
        
        for i in range(0, len(embeddings), batch_size):
            batch = embeddings[i:i+batch_size]
            labels = clusterer.fit_predict(batch)
            all_labels.extend(labels)
            
        # 4. Формируем кластеры
        clusters_dict = {}
        for idx, (chunk, label) in enumerate(zip(chunks, all_labels)):
            if label == -1:  # Шум - отдельные важные чанки
                label = f"singleton_{idx}"
            
            if label not in clusters_dict:
                clusters_dict[label] = {
                    'chunks': [],
                    'embeddings': [],
                    'size': 0
                }
            
            clusters_dict[label]['chunks'].append(chunk)
            clusters_dict[label]['embeddings'].append(embeddings[idx])
            clusters_dict[label]['size'] += 1
        
        # 5. Создаем объекты кластеров
        clusters = []
        for label, data in clusters_dict.items():
            if len(data['chunks']) > 1:  # Только группы
                centroid = np.mean(data['embeddings'], axis=0)
                
                # Находим репрезентативный чанк (ближайший к центроиду)
                distances = [cosine_similarity([centroid], [emb])[0][0] 
                           for emb in data['embeddings']]
                rep_idx = np.argmax(distances)
                
                # Извлекаем ключевые слова
                keywords = self.extract_keywords(data['chunks'])
                
                cluster = ChunkCluster(
                    id=f"cluster_{len(clusters)}",
                    name=f"Тематическая группа {len(clusters)}",
                    chunks=[c['id'] for c in data['chunks']],
                    centroid=centroid,
                    theme=keywords[0] if keywords else "unknown",
                    keywords=keywords,
                    size=len(data['chunks']),
                    representative_chunk=data['chunks'][rep_idx]
                )
                clusters.append(cluster)
        
        # Ограничиваем количество кластеров
        clusters.sort(key=lambda x: x.size, reverse=True)
        return clusters[:target_clusters]
    
    def extract_keywords(self, chunks: List[dict]) -> List[str]:
        """Извлечение ключевых слов из кластера чанков"""
        # Используем TF-IDF или KeyBERT для извлечения ключевых слов
        texts = [chunk['text'] for chunk in chunks if 'text' in chunk]
        # Здесь должна быть логика извлечения ключевых слов
        return ["ключевое_слово1", "ключевое_слово2"]
```

```javascript
        if (num >= 1000000) return (num/1000000).toFixed(1) + 'M';

            this.levels[l].visible = parseInt(l) >= this.currentLevel;

        const endpoints = {
            4: '/api/graph/concepts',        // Концепции (50 узлов)
            3: '/api/graph/components',       // Компоненты (500 узлов)
            2: '/api/graph/documents',        // Документы (5000 узлов)
            1: '/api/graph/topics',           // Темы (50000 узлов)
            0: '/api/graph/chunks'            // Чанки (1M+ узлов) - НЕ ИСПОЛЬЗУЕМ

        if (level === 0) {
            // Для уровня чанков возвращаем агрегированные данные
            return this.getAggregatedChunkData();

            this.graph._destructor();

        const config = this.getRenderConfig(this.currentLevel);
        
        this.graph = ForceGraph3D()(this.container)
            .graphData(data)
            .nodeRelSize(config.nodeSize)
            .nodeResolution(config.resolution)
            .linkWidth(config.linkWidth)
            .nodeLabel(node => this.getNodeLabel(node, this.currentLevel))
            .nodeColor(node => this.getNodeColor(node, this.currentLevel))
            .nodeVal(node => this.getNodeSize(node, this.currentLevel))
            .onNodeClick(node => this.handleNodeClick(node));

            4: { nodeSize: 8, resolution: 32, linkWidth: 1.5 },  // Концепции - детально
            3: { nodeSize: 5, resolution: 24, linkWidth: 1.0 },  // Компоненты
            2: { nodeSize: 3, resolution: 16, linkWidth: 0.8 },  // Документы
            1: { nodeSize: 2, resolution: 12, linkWidth: 0.5 },  // Темы - упрощенно
            0: { nodeSize: 1, resolution: 8, linkWidth: 0.3 }    // Кластеры - минимально
        };
        return configs[level] || configs[2];
    }

            case 4: return `${node.name}\n🏛️ Концепция\nСвязано: ${node.connections}`;
            case 3: return `${node.name}\n📦 Компонент\nДокументов: ${node.docCount}`;
            case 2: return `${node.title}\n📄 Документ\nЧанков: ${node.chunkCount}`;
            case 1: return `${node.theme}\n📑 Тема\nЧанков: ${node.size}`;

        const baseSizes = {4: 2.0, 3: 1.5, 2: 1.2, 1: 1.0, 0: 0.8};
        const base = baseSizes[level];
        
        // Размер зависит от количества связанных элементов
        switch(level) {
            case 4: return base * (1 + Math.log2(node.importance || 1) * 0.3);
            case 3: return base * (1 + Math.log2(node.docCount || 1) * 0.2);
            case 2: return base * (1 + Math.log2(node.chunkCount || 1) * 0.15);
            case 1: return base * (1 + Math.log2(node.size || 1) * 0.1);
            case 0: return base * (1 + Math.log2(node.size || 1) * 0.05);
        }
    }
    
    async handleNodeClick(node) {
        // При клике на узел - углубляемся

            { x: centerNode.x, y: centerNode.y, z: centerNode.z },
            2000
        );
    }
```

```python
                    "importance": record["importance"],
                    "hasChildren": True
                }
                nodes.append(node)
                
                # Связи между концепциями
                # ...
            
            return {"nodes": nodes, "links": links}
    
    async def get_topic_graph(self, limit: int = 1000):
        """
        Уровень 1: Тематические кластеры чанков
        ~50,000 тем (агрегировано до 1000)
        """
        async with self.driver.session() as session:
            # Получаем агрегированные данные по темам
            result = await session.run("""
                MATCH (t:Topic)<-[:BELONGS_TO]-(c:Chunk)
                WITH t, collect(c) as chunks, count(c) as chunk_count
                WHERE chunk_count > 5
                RETURN t.name as theme, 
                       chunk_count,
                       [chunk in chunks[0..5] | chunk.text] as examples,
                       t.embedding as centroid
                ORDER BY chunk_count DESC
                LIMIT $limit
            """, limit=limit)
            
            topics = []
            async for record in result:
                topic = {
                    "id": f"topic_{hash(record['theme'])}",
                    "theme": record["theme"],
                    "size": record["chunk_count"],
                    "centroid": record["centroid"],
                    "examples": record["examples"],
                    "type": "Topic"
                }
                topics.append(topic)
            
            # Вычисляем связи между темами (по близости эмбеддингов)
            embeddings = [t["centroid"] for t in topics]
            links = self.calculate_topic_links(embeddings, topics)
            
            return {"nodes": topics, "links": links}
    
    def calculate_topic_links(self, embeddings, topics, threshold=0.7):
        """
        Создание связей между близкими темами
        """
        links = []
        for i, emb_i in enumerate(embeddings):
            for j, emb_j in enumerate(embeddings[i+1:], i+1):
                similarity = np.dot(emb_i, emb_j) / (
                    np.linalg.norm(emb_i) * np.linalg.norm(emb_j)
                )
                if similarity > threshold:
                    links.append({
                        "source": topics[i]["id"],
                        "target": topics[j]["id"],
                        "type": "RELATED",
                        "strength": float(similarity)
                    })
        return links
    
    async def expand_node(self, node_id: str, current_level: int):
        """
        Получение деталей узла (для drill-down)
        """
        if current_level == 4:  # Концепция -> Компоненты
            return await self.get_components_for_concept(node_id)
        elif current_level == 3:  # Компонент -> Документы
            return await self.get_documents_for_component(node_id)
        elif current_level == 2:  # Документ -> Темы
            return await self.get_topics_for_document(node_id)
        elif current_level == 1:  # Тема -> Чанки
            return await self.get_chunks_for_topic(node_id)
        else:
            return {"nodes": [], "links": []}
```

Наблюдение: При длительном авто-вращении появляются микро-фризы. Нет возможности ручного управления режимом вращения.

**Проблема №3: Отсутствие визуальной иерархии**
Наблюдение: Все узлы одного размера, хотя логически Chunk (атомарные единицы) должны быть меньше, а Person/Technology/Component — крупнее.
