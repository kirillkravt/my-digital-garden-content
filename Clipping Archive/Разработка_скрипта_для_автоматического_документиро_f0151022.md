### 📝 Основной скрипт: `uch-create-note`

```bash
#!/bin/bash

# Конфигурация
VAULT_DIR="/Users/kirillkravcov/obsidian/my-digital-garden-content/uch-docs"
TEMPLATES_DIR="$VAULT_DIR/_templates"
ARCHIVE_DIR="$VAULT_DIR/_archive"
SCRIPTS_DIR="$VAULT_DIR/_scripts"

# Функции
get_next_hex_id() {
    local parent_id="$1"
    local level="$2"
    
    # Определяем уровень
    if [[ -z "$parent_id" ]]; then
        # Корневой уровень
        existing_files=$(find "$VAULT_DIR" -maxdepth 1 -name "00-*" -type f | sort)
        if [[ -z "$existing_files" ]]; then
            echo "00"
            return
        fi
        
        # Находим максимальный номер
        max_num=0
        for file in $existing_files; do
            filename=$(basename "$file")
            if [[ "$filename" =~ ^00-([0-9A-F]{2}) ]]; then
                num=$((16#${BASH_REMATCH[1]}))
                if (( num > max_num )); then
                    max_num=$num
                fi
            fi
        done
        
        # Следующий ID в hex
        next_num=$((max_num + 1))
        printf "00-%02X" $next_num
    else
        # Дочерний уровень
        parent_pattern=$(echo "$parent_id" | sed 's/-/\\-/g')
        existing_files=$(find "$VAULT_DIR" -name "${parent_id}-*" -type f | sort)
        
        if [[ -z "$existing_files" ]]; then
            echo "${parent_id}-01"
            return
        fi
        
        # Находим максимальный суффикс
        max_num=0
        for file in $existing_files; do
            filename=$(basename "$file")
            if [[ "$filename" =~ ^${parent_id}-([0-9A-F]{2}) ]]; then
                num=$((16#${BASH_REMATCH[1]}))
                if (( num > max_num )); then
                    max_num=$num
                fi
            fi
        done
        
        next_num=$((max_num + 1))
        echo "${parent_id}-$(printf "%02X" $next_num)"
    fi
}

update_parent_document() {
    local parent_id="$1"
    local child_id="$2"
    local child_name="$3"
    
    parent_file=$(find "$VAULT_DIR" -name "${parent_id}*.md" -type f | head -1)
    
    if [[ -f "$parent_file" ]]; then
        # Ищем раздел "ДОЧЕРНИЕ:" или создаем его
        if grep -q "#### ДОЧЕРНИЕ:" "$parent_file"; then
            # Вставляем ссылку в раздел ДОЧЕРНИЕ
            sed -i '' "/#### ДОЧЕРНИЕ:/a\\
- [[${child_id} - ${child_name}]] - ${child_name}" "$parent_file"
        elif grep -q "### СВЯЗАННЫЕ ДОКУМЕНТЫ:" "$parent_file"; then
            # Вставляем после СВЯЗАННЫЕ ДОКУМЕНТЫ
            sed -i '' "/### СВЯЗАННЫЕ ДОКУМЕНТЫ:/a\\
#### ДОЧЕРНИЕ:\\
- [[${child_id} - ${child_name}]] - ${child_name}" "$parent_file"
        else
            # Добавляем в конец файла
            echo -e "\n### СВЯЗАННЫЕ ДОКУМЕНТЫ:\n#### ДОЧЕРНИЕ:\n- [[${child_id} - ${child_name}]] - ${child_name}" >> "$parent_file"
        fi
    fi
}

create_document() {
    local parent_id="$1"
    local doc_name="$2"
    local tags="$3"
    
    # Получаем следующий ID
    if [[ -z "$parent_id" ]] || [[ "$parent_id" == "мастер" ]]; then
        # Документ уровня 1
        doc_id=$(get_next_hex_id "" 1)
        template="$TEMPLATES_DIR/master-template.md"
        level=1
    else
        # Проверяем уровень родителя
        parent_level=$(echo "$parent_id" | tr -cd '-' | wc -c)
        level=$((parent_level + 2))
        doc_id=$(get_next_hex_id "$parent_id" $level)
        template="$TEMPLATES_DIR/child-template.md"
    fi
    
    # Создаем имя файла
    filename="${doc_id} - ${doc_name}.md"
    filepath="$VAULT_DIR/$filename"
    
    # Копируем шаблон
    cp "$template" "$filepath"
    
    # Заполняем шаблон
    current_date=$(date +%Y-%m-%d)
    
    # Для дочерних документов
    if [[ $level -gt 1 ]]; then
        # Получаем имя родителя
        parent_file=$(find "$VAULT_DIR" -name "${parent_id}*.md" -type f | head -1)
        parent_name=$(basename "$parent_file" .md | sed 's/^[^-]*- //')
        
        sed -i '' "s/{{PARENT_ID}}/$parent_id/g" "$filepath"
        sed -i '' "s/{{PARENT_NAME}}/$parent_name/g" "$filepath"
        sed -i '' "s/{{DOC_ID}}/$doc_id/g" "$filepath"
        sed -i '' "s/{{DOC_NAME}}/$doc_name/g" "$filepath"
        sed -i '' "s/{{LEVEL}}/$level/g" "$filepath"
        sed -i '' "s/{{TAGS}}/$tags/g" "$filepath"
        sed -i '' "s/{{CREATED_DATE}}/$current_date/g" "$filepath"
    else
        # Для корневых документов
        sed -i '' "s/{{DOC_ID}}/$doc_id/g" "$filepath"
        sed -i '' "s/{{DOC_NAME}}/$doc_name/g" "$filepath"
        sed -i '' "s/{{TAGS}}/$tags/g" "$filepath"
        sed -i '' "s/{{CREATED_DATE}}/$current_date/g" "$filepath"
    fi
    
    # Обновляем родительский документ
    if [[ $level -gt 1 ]]; then
        update_parent_document "$parent_id" "$doc_id" "$doc_name"
    fi
    
    # Обновляем метрики
    "$SCRIPTS_DIR/uch-metrics-update"
    
    echo "✅ Создан документ: $filename"
    echo "📁 ID: $doc_id"
    echo "📊 Уровень: $level"
    
    # Открываем в VSCode
    code "$filepath"
}

# Основная логика
if [[ $# -lt 2 ]]; then
    echo "Использование:"
    echo "  Создание корневого документа:"
    echo "    $0 \"\" \"Название документа\" \"@тег1 @тег2\""
    echo "  Создание дочернего документа:"
    echo "    $0 \"Родительский-ID\" \"Название документа\" \"@тег1 @тег2\""
    exit 1
fi

parent_id="$1"
doc_name="$2"
tags="${3:-@active}"

create_document "$parent_id" "$doc_name" "$tags"
```

### 📦 Скрипт архивации: `uch-archive-note`

```bash
#!/bin/bash

VAULT_DIR="/Users/kirillkravcov/obsidian/my-digital-garden-content/uch-docs"
ARCHIVE_DIR="$VAULT_DIR/_archive"

archive_document() {
    local doc_id="$1"
    local reason="${2:-Archived}"
    
    # Находим файл
    file=$(find "$VAULT_DIR" -name "${doc_id}*.md" -type f | head -1)
    
    if [[ ! -f "$file" ]]; then
        echo "❌ Документ не найден: $doc_id"
        return 1
    fi
    
    # Создаем структуру архива
    current_date=$(date +%Y-%m-%d)
    archive_path="$ARCHIVE_DIR/$current_date"
    mkdir -p "$archive_path"
    
    # Сохраняем с метаданными
    filename=$(basename "$file")
    archived_file="$archive_path/$filename"
    
    # Добавляем информацию об архивации
    echo -e "\n---\n### АРХИВАЦИЯ\n- **Дата архивации**: $current_date\n- **Причина**: $reason\n- **Архивный путь**: $archive_path\n---\n" >> "$file"
    
    # Копируем в архив
    cp "$file" "$archived_file"
    
    # Обновляем статус в оригинале
    sed -i '' "s/**Статус**:.*/**Статус**: 📦 АРХИВИРОВАН ($current_date)/g" "$file"
    
    # Удаляем из родительских ссылок
    # (дополнительная функция для поиска всех ссылок)
    
    echo "✅ Документ архивирован: $filename"
    echo "📦 Архив: $archive_path"
}

# Основная логика
if [[ $# -lt 1 ]]; then
    echo "Использование: $0 <ID-документа> [причина]"
    exit 1
fi

archive_document "$1" "$2"
```

### 📊 Скрипт обновления метрик: `uch-metrics-update`

```bash
#!/bin/bash

VAULT_DIR="/Users/kirillkravcov/obsidian/my-digital-garden-content/uch-docs"

update_metrics() {
    # Считаем документы
    total_docs=$(find "$VAULT_DIR" -name "*.md" -type f | grep -v "_archive\|_templates\|_scripts" | wc -l)
    
    # Считаем активные задачи
    active_tasks=$(grep -r "@active" "$VAULT_DIR" --include="*.md" | wc -l)
    
    # Считаем завершенные
    completed_tasks=$(grep -r "@done" "$VAULT_DIR" --include="*.md" | wc -l)
    
    # Максимальный уровень
    max_level=0
    for file in "$VAULT_DIR"/*.md; do
        if [[ -f "$file" ]]; then
            filename=$(basename "$file")
            if [[ "$filename" =~ ^00 ]]; then
                level=$(echo "$filename" | tr -cd '-' | wc -c)
                if (( level > max_level )); then
                    max_level=$level
                fi
            fi
        fi
    done
    
    # Обновляем главный документ
    master_file="$VAULT_DIR/00.md"
    if [[ -f "$master_file" ]]; then
        sed -i '' "s/**Всего документов**:.*/**Всего документов**: $total_docs/g" "$master_file"
        
        # Добавляем дополнительные метрики
        if grep -q "### МЕТРИКИ" "$master_file"; then
            sed -i '' "/### МЕТРИКИ/a\\
- **Активные задачи**: $active_tasks\\
- **Завершено**: $completed_tasks\\
- **Максимальный уровень**: $max_level\\
- **Последнее обновление**: $(date +%Y-%m-%d)" "$master_file"
        fi
    fi
    
    echo "📊 Метрики обновлены:"
    echo "   Всего документов: $total_docs"
    echo "   Активные задачи: $active_tasks"
    echo "   Завершено: $completed_tasks"
    echo "   Максимальный уровень: $max_level"
}

update_metrics
```

```bash
#!/bin/bash
# uch-create-note-enhanced.sh

# Функция для создания YAML frontmatter
create_frontmatter() {
    local type="$1"
    local level="$2"
    local id="$3"
    local name="$4"
    local parent="$5"
    local parent_name="$6"
    local tags="$7"
    local author="$8"
    
    case "$type" in
        "master")
            cat << YAML
---
id: "$id"
name: "$name"
type: "project"
level: 1
status: "active-development"
version: "0.1.0"
tags: [$tags, "@project", "@master"]
parent: null
children: []
related: []
created: "$(date +%Y-%m-%d)"
updated: "$(date +%Y-%m-%d)"
priority: "medium"
author: "$author"
---
YAML
            ;;
        "child")
            cat << YAML
---
id: "$id"
name: "$name"
type: "component"
level: $level
status: "in-development"
tags: [$tags, "@component", "@child"]
parent: "$parent"
parent_name: "$parent_name"
children: []
related: []
created: "$(date +%Y-%m-%d)"
updated: "$(date +%Y-%m-%d)"
priority: "medium"
author: "$author"
due_date: null
depends_on: []
blocks: []
---
YAML
            ;;
        "task")
            cat << YAML
---
id: "$id"
name: "$name"
type: "task"
level: $level
status: "open"
tags: [$tags, "@task"]
parent: "$parent"
parent_name: "$parent_name"
component: "$(echo $parent | cut -d'-' -f1-3)"
assignee: "$author"
created: "$(date +%Y-%m-%d)"
updated: "$(date +%Y-%m-%d)"
due_date: "$(date -v+7d +%Y-%m-%d)"
priority: "medium"
estimate: "4"
time_spent: 0
depends_on: []
blocks: []
---
YAML
            ;;
    esac
}

# Функция обновления родительского документа
update_parent_yaml() {
    local parent_file="$1"
    local child_id="$2"
    
    if [[ -f "$parent_file" ]]; then
        # Извлекаем текущий YAML
        yaml_content=$(awk '/^---$/{if(++count==2) exit} count' "$parent_file")
        
        # Обновляем поле children
        if echo "$yaml_content" | grep -q "children:"; then
            # Добавляем child_id в массив children
            sed -i '' "/children:/a\  - \"$child_id\"" "$parent_file"
        fi
        
        # Также обновляем текстовый раздел для совместимости
        if grep -q "#### ДОЧЕРНИЕ:" "$parent_file"; then
            sed -i '' "/#### ДОЧЕРНИЕ:/a\\
- [[$child_id]]" "$parent_file"
        fi
    fi
}

# Основная логика создания документа
create_document_with_yaml() {
    local parent_id="$1"
    local doc_name="$2"
    local tags="$3"
    local author="${4:-$(whoami)}"
    
    # Определяем тип и уровень
    if [[ -z "$parent_id" ]]; then
        type="master"
        level=1
        doc_id=$(get_next_hex_id "" $level)
        parent_name=""
    else
        parent_level=$(echo "$parent_id" | tr -cd '-' | wc -c)
        level=$((parent_level + 2))
        doc_id=$(get_next_hex_id "$parent_id" $level)
        
        if [[ $level -eq 4 ]]; then
            type="task"
        else
            type="child"
        fi
        
        # Получаем имя родителя
        parent_file=$(find "$VAULT_DIR" -name "${parent_id}*.md" -type f | head -1)
        parent_name=$(basename "$parent_file" .md | sed 's/^[^-]*- //')
    fi
    
    # Создаем frontmatter
    frontmatter=$(create_frontmatter "$type" "$level" "$doc_id" "$doc_name" \
        "$parent_id" "$parent_name" "$tags" "$author")
    
    # Создаем файл
    filename="${doc_id} - ${doc_name}.md"
    filepath="$VAULT_DIR/$filename"
    
    # Записываем frontmatter и выбираем шаблон
    echo "$frontmatter" > "$filepath"
    
    # Добавляем контент из шаблона (без frontmatter)
    template_file="$TEMPLATES_DIR/${type}-template.md"
    if [[ -f "$template_file" ]]; then
        # Пропускаем frontmatter в шаблоне (между ---)
        awk '/^---$/{if(++count==2) next} count<2{next} {print}' "$template_file" >> "$filepath"
    fi
    
    # Обновляем родительский документ
    if [[ -n "$parent_id" ]]; then
        update_parent_yaml "$parent_file" "$doc_id"
    fi
    
    echo "✅ Создан документ: $filename"
    echo "📄 Frontmatter:"
    echo "$frontmatter" | head -20
}
```

```bash
#!/bin/bash
# create-master-v2.sh - с автонумерацией

VAULT_DIR="/Users/kirillkravcov/obsidian/my-digital-garden-content/uch-docs"

# Функция получения следующего HEX ID
get_next_master_id() {
    # Находим все мастер-файлы (уровень 1)
    existing_ids=$(find "$VAULT_DIR" -maxdepth 1 -name "*.md" -type f | \
        grep -E '^[0-9A-F]{2} - ' | \
        sed 's/.*\/\([0-9A-F]\{2\}\) -.*/\1/' | \
        sort)
    
    if [[ -z "$existing_ids" ]]; then
        echo "00"
        return
    fi
    
    # Конвертируем в десятичные, находим максимальный
    max_dec=0
    for hex_id in $existing_ids; do
        dec_id=$((16#$hex_id))
        if (( dec_id > max_dec )); then
            max_dec=$dec_id
        fi
    done
    
    # Следующий ID в hex
    next_dec=$((max_dec + 1))
    printf "%02X" $next_dec
}

# Основная логика
echo "=== СОЗДАНИЕ МАСТЕР-ДОКУМЕНТА v2 ==="

# Автоматически генерируем ID
auto_id=$(get_next_master_id)
echo "Сгенерирован ID: $auto_id"

read -p "Введите название проекта: " doc_name
read -p "Введите теги (через пробел, без @): " doc_tags

# Форматируем теги
formatted_tags=""
for tag in $doc_tags; do
    formatted_tags="$formatted_tags \"@$tag\","
done
formatted_tags="${formatted_tags%,}"  # Убираем последнюю запятую

current_date=$(date +%Y-%m-%d)
filename="${auto_id} - ${doc_name}.md"
full_path="$VAULT_DIR/$filename"

echo "Создаю: $filename"
echo "Путь: $full_path"

# Создаем документ
cat > "$full_path" << EOF
---
id: "$auto_id"
name: "$doc_name"
type: "project"
level: 1
status: "active-development"
version: "0.1.0"
tags: [$formatted_tags]
parent: null
children: []
related: []
created: "$current_date"
updated: "$current_date"
priority: "medium"
author: "$USER"
---
# $doc_name

## 📊 ОБЩАЯ ИНФОРМАЦИЯ
```

## **ШАГ 4: Добавляем автонумерацию HEX ID**

### **4.1. Действие: Создаем скрипт с автонумерацией**

**Содержимое файла:**
```bash
#!/bin/bash

echo "=== СОЗДАНИЕ МАСТЕР-ДОКУМЕНТА С АВТОНОМЕРАЦИЕЙ ==="

# Функция для получения следующего HEX ID
get_next_hex_id() {
    # Ищем все файлы с паттерном ID в начале имени
    # Паттерн: две hex цифры, пробел, дефис, пробел
    existing_ids=$(find . -maxdepth 1 -name "*.md" -type f | \
        grep -E '^[0-9A-Fa-f]{2} - ' | \
        sed 's/\.\///' | \
        cut -d' ' -f1 | \
        sort | uniq)
    
    # Если нет файлов, начинаем с 00
    if [ -z "$existing_ids" ]; then
        echo "00"
        return
    fi
    
    # Находим максимальный ID
    max_id=0
    for id in $existing_ids; do
        # Конвертируем hex в decimal
        dec_id=$((16#$id))
        if [ $dec_id -gt $max_id ]; then
            max_id=$dec_id
        fi
    done
    
    # Увеличиваем на 1 и конвертируем обратно в hex
    next_id=$((max_id + 1))
    printf "%02X" $next_id
}
```

## **ШАГ 5: Исправляем логику автонумерации - ищем первый свободный ID**

```bash
# Функция для поиска первого свободного HEX ID
find_first_free_id() {
    # Получаем список всех ID в текущей папке
    # Ищем файлы с паттерном: две hex цифры, пробел, дефис, пробел
    existing_ids=$(find . -maxdepth 1 -name "*.md" -type f 2>/dev/null | \
        grep -E '^\./[0-9A-Fa-f]{2} - ' | \
        sed 's/^\.\///' | \
```

```bash
#!/bin/bash

echo "=== КОНВЕРТАЦИЯ СУЩЕСТВУЮЩИХ ДОКУМЕНТОВ В ШАБЛОН ==="

# Функция для извлечения ID и имени из имени файла
extract_id_and_name() {
    local filename="$1"
    
    # Пытаемся извлечь ID (первые 2, 5, 8, 11 символов до дефиса)
    if [[ "$filename" =~ ^([0-9A-Fa-f]{2})(-[0-9A-Fa-f]{2})* ]]; then
        id="${BASH_REMATCH[0]}"
        # Убираем ID и " - " из начала имени
        name=$(echo "$filename" | sed -E "s/^${id} - //" | sed 's/\.md$//')
        echo "$id|$name"
    else
        echo ""
    fi
}

# Функция для определения уровня по ID
get_level_from_id() {
    local id="$1"
    if [ -z "$id" ]; then
        echo "0"
        return
    fi
    # Считаем количество дефисов в ID
    level=$(echo "$id" | tr -cd '-' | wc -c)
    echo $((level + 1))
}

# Функция для определения типа документа по уровню
get_type_from_level() {
    local level="$1"
    case $level in
        1) echo "project" ;;
        2|3) echo "component" ;;
        *) echo "task" ;;
    esac
}

# Функция для конвертации одного файла
convert_file() {
    local file="$1"
    local dry_run="${2:-false}"
    
    echo "Обрабатываю: $file"
    
    # Извлекаем ID и имя из имени файла
    id_name=$(extract_id_and_name "$(basename "$file")")
    if [ -z "$id_name" ]; then
        echo "  ⚠️  Пропускаю: не удалось извлечь ID из имени файла"
        return
    fi
    
    id=$(echo "$id_name" | cut -d'|' -f1)
    name=$(echo "$id_name" | cut -d'|' -f2)
    level=$(get_level_from_id "$id")
    type=$(get_type_from_level "$level")
    
    echo "  ID: $id, Уровень: $level, Тип: $type"
    
    # Проверяем, есть ли уже frontmatter в файле
    if head -1 "$file" | grep -q "^---"; then
        echo "  ℹ️  Файл уже имеет frontmatter, пропускаю"
        return
    fi
    
    # Читаем существующее содержимое
    existing_content=$(cat "$file")
    current_date=$(date +%Y-%m-%d)
    
    if [ "$dry_run" = "true" ]; then
        echo "  📝 Будет добавлен frontmatter:"
        echo "  ---"
        echo "  id: \"$id\""
        echo "  name: \"$name\""
        echo "  type: \"$type\""
        echo "  level: $level"
        echo "  status: \"active\""
        echo "  tags: [\"@$type\"]"
        echo "  created: \"$current_date\""
        echo "  updated: \"$current_date\""
        echo "  author: \"$USER\""
        echo "  ---"
        return
    fi
    
    # Создаем временный файл с frontmatter
    temp_file="${file}.tmp"
    
    cat > "$temp_file" << EOF
---
id: "$id"
name: "$name"
type: "$type"
level: $level
status: "active"
tags: ["@$type"]
created: "$current_date"
updated: "$current_date"
author: "$USER"
---

$existing_content
EOF
    
    # Заменяем оригинальный файл
    mv "$temp_file" "$file"
    
    echo "  ✅ Конвертирован"
}

# Основная логика
echo "Режим работы:"
echo "1 - Только показать что будет сделано (dry run)"
echo "2 - Выполнить конвертацию"
read -p "Выберите режим (1 или 2): " mode

dry_run="false"
if [ "$mode" = "1" ]; then
    dry_run="true"
    echo "Режим: Только предпросмотр"
elif [ "$mode" = "2" ]; then
    dry_run="false"
    echo "Режим: Выполнение конвертации"
else
    echo "Ошибка: неверный выбор"
    exit 1
fi

# Ищем все .md файлы с ID в начале имени
echo ""
echo "Поиск документов для конвертации..."
files=$(find . -maxdepth 1 -name "*.md" -type f | \
    grep -E '^\./[0-9A-Fa-f]{2}(-[0-9A-Fa-f]{2})* - ' | \

if [ -z "$files" ]; then
    echo "Не найдено документов для конвертации"
    exit 0
fi

echo "Найдено документов: $(echo "$files" | wc -l)"

count=0
for file in $files; do

    echo ""
    count=$((count + 1))
done

if [ "$dry_run" = "false" ]; then
    echo "✅ Конвертировано документов: $count"
else
    echo "📋 Будет конвертировано документов: $count"
fi
```

```bash
# Функция для добавления ссылки в родительский документ
update_parent_document() {
    local parent_file="$1"
    local child_id="$2"
    local child_name="$3"
    
    echo "Обновляю родительский документ: $parent_file"
    
    if [ ! -f "$parent_file" ]; then
        echo "Ошибка: родительский файл не найден: $parent_file"
        return 1
    fi
    
    # Создаем временный файл
    temp_file="${parent_file}.tmp"
    
    # Удаляем строку "Пока нет дочерних документов" если она есть
    # и добавляем/обновляем раздел ДОЧЕРНИЕ ДОКУМЕНТЫ
    awk -v child_id="$child_id" -v child_name="$child_name" '
    BEGIN { in_children_section = 0; children_section_found = 0; children_added = 0 }
    
    # Если находим начало раздела ДОЧЕРНИЕ ДОКУМЕНТЫ
    /#### ДОЧЕРНИЕ ДОКУМЕНТЫ/ {
        in_children_section = 1
        children_section_found = 1
        print $0
        next
    }
    
    # Если мы внутри раздела и находим "Пока нет дочерних документов" - пропускаем
    in_children_section && /^[[:space:]]*Пока нет дочерних документов/ {
        # Пропускаем эту строку
        next
    }
    
    # Если мы внутри раздела и находим пустую строку или новый раздел - выходим из раздела
    in_children_section && (/^####/ || /^###/ || /^##/ || /^#/ || /^---/) {
        in_children_section = 0
    }
    
    # Если мы внутри раздела и это конец файла
    in_children_section && /^[[:space:]]*$/ {
        # Добавляем ссылку перед пустой строкой
        print "- [[" child_id " - " child_name "]]"
        in_children_section = 0
        children_added = 1
        print $0
        next
    }
    
    # Если нашли раздел ДОЧЕРНИЕ ДОКУМЕНТЫ, но еще не добавили ссылку
    children_section_found && !children_added && !in_children_section && /^[^-]/ {
        # Добавляем ссылку после раздела
        print "- [[" child_id " - " child_name "]]"
        children_added = 1
        print $0
        next
    }
    
    { print $0 }
    
    END {
        # Если раздел не найден, добавляем его перед ---
        if (!children_section_found) {
            print ""
            print "#### ДОЧЕРНИЕ ДОКУМЕНТЫ"
            print "- [[" child_id " - " child_name "]]"
        }
    }
    ' "$parent_file" > "$temp_file"
    
    # Заменяем оригинальный файл
    mv "$temp_file" "$parent_file"
    
    echo "✅ Ссылка добавлена в родительский документ"
}
```

```bash
#!/bin/bash
```

```bash
# Функция для поиска первого свободного ID для мастер-документов
find_free_master_id() {
    # Ищем все мастер-файлы (уровень 1)
    existing_ids=$(find . -maxdepth 1 -name "*.md" -type f 2>/dev/null | \
        grep -E '^\./[0-9A-Fa-f]{2} - ' | \
        sed 's/^\.\///' | \
        cut -d' ' -f1 | \
        sort | uniq)
    
    # Если нет файлов, начинаем с 00
    if [ -z "$existing_ids" ]; then
        echo "00"
        return
    fi
    
    # Конвертируем все ID в десятичные
    declare -a dec_ids=()
    for hex_id in $existing_ids; do
        dec_id=$((16#$hex_id))
        dec_ids+=($dec_id)
    done
    
    # Сортируем и ищем первый пропуск
    sorted_ids=($(printf "%d\n" "${dec_ids[@]}" | sort -n))
    expected=0
    for id in "${sorted_ids[@]}"; do
        if [ $id -gt $expected ]; then
            break
        fi
        expected=$((id + 1))
    done
    
    printf "%02X" $expected
}

# Функция для поиска первого свободного дочернего ID
find_free_child_id() {
    local parent_id="$1"
    
    # Ищем все файлы, которые начинаются с parent_id-
    pattern="${parent_id}-[0-9A-Fa-f][0-9A-Fa-f]"
    
    # Получаем список всех дочерних ID
    existing_ids=$(find . -maxdepth 1 -name "*.md" -type f 2>/dev/null | \
        grep -E "^\./${pattern} - " | \
        sed 's/^\.\///' | \
        cut -d' ' -f1 | \
        grep -E "^${parent_id}-[0-9A-Fa-f]{2}$" | \
        awk -F"${parent_id}-" '{print $2}' | \
        sort | uniq)
    
    # Если нет дочерних файлов, начинаем с 01
    if [ -z "$existing_ids" ]; then
        echo "01"
        return
    fi
    
    # Конвертируем hex ID в decimal и находим пропуски
    declare -a dec_array=()
    for hex_id in $existing_ids; do
        dec_id=$((16#$hex_id))
        dec_array+=($dec_id)
    done
    
    # Сортируем массив чисел
    sorted_array=($(printf "%d\n" "${dec_array[@]}" | sort -n))
    
    # Ищем первый пропуск, начиная с 1
    expected=1
    for id in "${sorted_array[@]}"; do
        if [ $id -gt $expected ]; then
            break
        fi
        expected=$((id + 1))
    done
    
    printf "%02X" $expected
}

# Функция для обновления родительского документа
update_parent_document() {
    local parent_file="$1"
    local child_id="$2"
    local child_name="$3"
    
    if [ ! -f "$parent_file" ]; then
        echo "  ⚠️  Родительский файл не найден, пропускаю обновление"
        return 1
    fi
    
    # Создаем временный файл
    temp_file="${parent_file}.tmp"
    
    # Удаляем "Пока нет дочерних документов" и добавляем ссылку
    awk -v child_id="$child_id" -v child_name="$child_name" '
    BEGIN { in_children_section = 0; children_section_found = 0; children_added = 0 }
    
    # Если находим начало раздела ДОЧЕРНИЕ ДОКУМЕНТЫ
    /#### ДОЧЕРНИЕ ДОКУМЕНТЫ/ {
        in_children_section = 1
        children_section_found = 1
        print $0
        next
    }
    
    # Если мы внутри раздела и находим "Пока нет дочерних документов" - пропускаем
    in_children_section && /^[[:space:]]*Пока нет дочерних документов/ {
        next
    }
    
    # Если мы внутри раздела и находим новый раздел - выходим из раздела
    in_children_section && (/^####/ || /^###/ || /^##/ || /^#/ || /^---/) {
        in_children_section = 0
    }
    
    # Если мы внутри раздела и это конец файла или пустая строка
    in_children_section && (/^[[:space:]]*$/ || /^---/) {
        # Добавляем ссылку перед пустой строкой или разделителем
        print "- [[" child_id " - " child_name "]]"
        in_children_section = 0
        children_added = 1
        print $0
        next
    }
    
    # Если нашли раздел ДОЧЕРНИЕ ДОКУМЕНТЫ, но еще не добавили ссылку
    children_section_found && !children_added && !in_children_section {
        # Добавляем ссылку после раздела
        print "- [[" child_id " - " child_name "]]"
        children_added = 1
        print $0
        next
    }
    
    { print $0 }
    
    END {
        # Если раздел не найден, добавляем его
        if (!children_section_found) {
            print ""
            print "#### ДОЧЕРНИЕ ДОКУМЕНТЫ"
            print "- [[" child_id " - " child_name "]]"
        }
    }
    ' "$parent_file" > "$temp_file"
    
    # Заменяем оригинальный файл
    mv "$temp_file" "$parent_file"
}
```

```bash
clean_parent_name() {
    local full_filename="$1"
    
    # Убираем расширение .md
    filename=$(basename "$full_filename" .md)
    
    echo "  Очистка имени из: $filename" >&2
    
    # Разделяем по " - " и берем все части кроме первой (ID)
    # Пример: "00-01 - 00-01 - Компонент" -> ["00-01", "00-01", "Компонент"]
    
    # Используем IFS для разделения
    OLD_IFS="$IFS"
    IFS=" - "
    parts=($filename)
    IFS="$OLD_IFS"
    
    # Проверяем сколько частей получилось
    if [ ${#parts[@]} -eq 0 ]; then
        echo "$filename"
        return
    fi
    
    # Первая часть - это всегда ID (или первый ID если их несколько)
    # Нам нужно проверить все части и оставить только те, которые НЕ являются ID
    
    clean_parts=()
    for part in "${parts[@]}"; do
        # Проверяем, является ли часть ID (hex формат с дефисами)
        if echo "$part" | grep -qE '^[0-9A-Fa-f]{2}(-[0-9A-Fa-f]{2})*$'; then
            # Это ID - пропускаем
            continue
        else
            # Это не ID - добавляем в результат
            clean_parts+=("$part")
        fi
    done
    
    # Если после очистки ничего не осталось, возвращаем последний ID как имя
    if [ ${#clean_parts[@]} -eq 0 ]; then
        # Берем последнюю часть (последний ID) как имя
        last_part="${parts[-1]}"
        echo "$last_part"
    else
        # Собираем оставшиеся части обратно в строку
        clean_name=$(IFS=" - "; echo "${clean_parts[*]}")
        echo "$clean_name"
```

```bash
#!/bin/bash

echo "=== ИСПРАВЛЕНИЕ ВСЕХ ССЫЛОК В ДОКУМЕНТАХ ==="

# Функция очистки имени (такая же как в основном скрипте)
clean_filename() {
    local filename="$1"
    
    # Убираем расширение .md
    filename=$(echo "$filename" | sed 's/\.md$//')
    
    # Разделяем по " - "
    OLD_IFS="$IFS"
    IFS=" - "
    parts=($filename)
    IFS="$OLD_IFS"
    
    if [ ${#parts[@]} -eq 0 ]; then
        echo "$filename"
        return
    fi
    
    clean_parts=()
    for part in "${parts[@]}"; do
        if echo "$part" | grep -qE '^[0-9A-Fa-f]{2}(-[0-9A-Fa-f]{2})*$'; then
            continue
        else
            clean_parts+=("$part")
        fi
    done
    
    if [ ${#clean_parts[@]} -eq 0 ]; then
        echo "${parts[-1]}"
    else
        clean_name=$(IFS=" - "; echo "${clean_parts[*]}")
        echo "$clean_name"
    fi
}

# Функция для исправления одного файла
fix_file_references() {
    local file="$1"
    local dry_run="$2"
    
    echo "Проверяю файл: $(basename "$file")"
    
    # Проверяем, есть ли в файле frontmatter с parent_name
    if grep -q "^parent_name:" "$file"; then
        # Извлекаем parent_id и текущее parent_name
        parent_id=$(grep "^parent:" "$file" | head -1 | sed 's/parent: *"//' | sed 's/"//')
        current_parent_name=$(grep "^parent_name:" "$file" | head -1 | sed 's/parent_name: *"//' | sed 's/"//')
        
        if [ -n "$parent_id" ]; then
            # Находим родительский файл
            parent_file=$(find . -maxdepth 1 -name "${parent_id} - *.md" -type f | head -1)
            
            if [ -n "$parent_file" ]; then
                # Получаем правильное имя родителя
                correct_parent_name=$(clean_filename "$(basename "$parent_file")")
                
                # Сравниваем с текущим
                if [ "$current_parent_name" != "$correct_parent_name" ]; then
                    echo "  ❌ Некорректное parent_name: '$current_parent_name'"
                    echo "  ✅ Правильное: '$correct_parent_name'"
                    
                    if [ "$dry_run" = "false" ]; then
                        # Исправляем parent_name в frontmatter
                        sed -i '' "s/^parent_name:.*$/parent_name: \"$correct_parent_name\"/" "$file"
                        echo "  ✅ Исправлен parent_name"
                        
                        # Также исправляем ссылку в тексте
                        if grep -q "\[\[$parent_id - " "$file"; then
                            sed -i '' "s/\[\[$parent_id - [^]]*\]\]/[[$parent_id - $correct_parent_name]]/" "$file"
                            echo "  ✅ Исправлена ссылка в тексте"
                        fi
                    fi
                else
                    echo "  ✅ parent_name корректно: '$current_parent_name'"
                fi
            fi
        fi
    fi
    
    # Также проверяем и исправляем ссылки в разделе РОДИТЕЛЬ
    if grep -q "#### ОБЩАЯ ИНФОРМАЦИЯ" "$file"; then
        # Ищем строку с Родитель:
        if grep -q "^- \*\*Родитель\*\*:" "$file" || grep -q "^Родитель:" "$file"; then
            # Находим родительскую ссылку в тексте
            if line=$(grep -n "\[\[.*\]\]" "$file" | grep -i "родитель" | head -1); then
                line_num=$(echo "$line" | cut -d: -f1)
                line_text=$(echo "$line" | cut -d: -f2-)
                
                # Извлекаем текущую ссылку
                if [[ "$line_text" =~ \[\[([^-]+) - ([^]]+)\]\] ]]; then
                    current_ref_id="${BASH_REMATCH[1]}"
                    current_ref_name="${BASH_REMATCH[2]}"
                    
                    # Находим правильное имя
                    parent_file=$(find . -maxdepth 1 -name "${current_ref_id} - *.md" -type f | head -1)
                    if [ -n "$parent_file" ]; then
                        correct_name=$(clean_filename "$(basename "$parent_file")")
                        
                        if [ "$current_ref_name" != "$correct_name" ]; then
                            echo "  ❌ Некорректная ссылка: '$current_ref_name'"
                            echo "  ✅ Правильная: '$correct_name'"
                            
                            if [ "$dry_run" = "false" ]; then
                                sed -i '' "${line_num}s/\[\[$current_ref_id - [^]]*\]\]/[[$current_ref_id - $correct_name]]/" "$file"
                                echo "  ✅ Исправлена текстовая ссылка"
                            fi
                        fi
                    fi
                fi
            fi
        fi
    fi
    
    echo ""
}
```

```bash
# Функция для добавления ссылки в родительский документ (ИСПРАВЛЕННАЯ)
update_parent_document() {
    local parent_file="$1"
    local child_id="$2"
    local child_name="$3"
    
    echo "  Обновляю родительский документ: $(basename "$parent_file")"
    
    if [ ! -f "$parent_file" ]; then
        echo "  ⚠️  Родительский файл не найден, пропускаю обновление"
        return 1
    fi
    
    # Создаем временный файл
    temp_file="${parent_file}.tmp"
    
    # Флаг для отслеживания, добавили ли мы ссылку
    link_added=0
    
    # Читаем файл построчно и обрабатываем
    awk -v child_id="$child_id" -v child_name="$child_name" '
    BEGIN { 
        in_children_section = 0
        children_section_found = 0
        link_exists = 0
        link_added = 0
    }
    
    # Проверяем, не существует ли уже такая ссылка
    /\[\[.* - .*\]\]/ && in_children_section {
        if ($0 ~ "\\[\\[" child_id " - " child_name "\\]\\]") {
            link_exists = 1
            print "  ℹ️  Ссылка уже существует, пропускаю" > "/dev/stderr"
        }
    }
    
    # Если находим начало раздела ДОЧЕРНИЕ ДОКУМЕНТЫ
    /#### ДОЧЕРНИЕ ДОКУМЕНТЫ/ {
        in_children_section = 1
        children_section_found = 1
        print $0
        next
    }
    
    # Если мы внутри раздела и находим "Пока нет дочерних документов" - пропускаем
    in_children_section && /^[[:space:]]*Пока нет дочерних документов/ {
        # Пропускаем эту строку
        next
    }
    
    # Если мы внутри раздела и находим новый раздел - выходим из раздела
    in_children_section && (/^####/ || /^###/ || /^##/ || /^#/ || /^---/) {
        in_children_section = 0
    }
    
    # Если мы внутри раздела и это пустая строка
    in_children_section && /^[[:space:]]*$/ {
        # Если ссылка еще не добавлена и не существует
        if (!link_exists && !link_added) {
            print "- [[" child_id " - " child_name "]]"
            link_added = 1
        }
        in_children_section = 0
        print $0
        next
    }
    
    # Если мы внутри раздела и это конец файла
    in_children_section && /^---/ {
        # Если ссылка еще не добавлена и не существует
        if (!link_exists && !link_added) {
            print "- [[" child_id " - " child_name "]]"
            link_added = 1
        }
        in_children_section = 0
        print $0
        next
    }
    
    # Если мы в конце раздела ДОЧЕРНИЕ ДОКУМЕНТЫ (последняя строка раздела)
    in_children_section && !/^- \[\[/ && !/^[[:space:]]*$/ {
        # Если это не ссылка и не пустая строка, значит конец списка ссылок
        # Добавляем новую ссылку перед выходом из раздела
        if (!link_exists && !link_added) {
            print "- [[" child_id " - " child_name "]]"
            link_added = 1
        }
        in_children_section = 0
        print $0
        next
    }
    
    # Печатаем текущую строку
    { print $0 }
    
    END {
        # Если раздел не найден, добавляем его
        if (!children_section_found) {
            print ""
            print "#### ДОЧЕРНИЕ ДОКУМЕНТЫ"
            print "- [[" child_id " - " child_name "]]"
        } else if (!link_exists && !link_added) {
            # Если раздел найден, но ссылка не добавлена (случай когда нет пустой строки в конце)
            print "- [[" child_id " - " child_name "]]"
        }
    }
    ' "$parent_file" > "$temp_file"
    
    # Заменяем оригинальный файл
    mv "$temp_file" "$parent_file"
    
    echo "  ✅ Ссылка добавлена в родительский документ"
}
```

**Содержимое файла:**
```bash
#!/bin/bash

echo "=== UCH CONVERT-FIX: ИСПРАВЛЕНИЕ ДОКУМЕНТОВ И ССЫЛОК ==="

# Функция для извлечения ID из имени файла
extract_id_from_filename() {
    local filename="$1"
    
    if [[ "$filename" =~ ^([0-9A-Fa-f]{2}(-[0-9A-Fa-f]{2})*) ]]; then
        echo "${BASH_REMATCH[1]}"
    else
        echo ""
    fi
}

# Функция для определения уровня по ID
get_level_from_id() {
    local id="$1"
    if [ -z "$id" ]; then
        echo "0"
        return
    fi
    level=$(echo "$id" | tr -cd '-' | wc -c)
    echo $((level + 1))
}

# Функция для поиска дочерних документов
find_child_documents() {
    local parent_id="$1"
    
    if [ -z "$parent_id" ]; then
        echo ""
        return
    fi
    
    # Ищем все файлы, которые начинаются с parent_id-
    child_files=$(find . -maxdepth 1 -name "${parent_id}-*.md" -type f 2>/dev/null | sort)
    echo "$child_files"
}

# Функция для поиска родительского документа
find_parent_document() {
    local child_id="$1"
    
    if [ -z "$child_id" ]; then
        echo ""
        return
    fi
    
    # Определяем parent_id (убираем последний сегмент)
    parent_id=$(echo "$child_id" | sed 's/-[^-]*$//')
    
    # Ищем родительский файл
    parent_file=$(find . -maxdepth 1 -name "${parent_id} - *.md" -type f | head -1)
    echo "$parent_file"
}

# Функция для добавления/обновления раздела ДОЧЕРНИЕ ДОКУМЕНТЫ
update_children_section() {
    local file="$1"
    local fix_mode="$2"
    
    echo "  Проверка раздела ДОЧЕРНИЕ ДОКУМЕНТЫ в: $(basename "$file")"
    
    # Извлекаем ID файла
    filename=$(basename "$file" .md)
    file_id=$(extract_id_from_filename "$filename")
    
    if [ -z "$file_id" ]; then
        echo "    ⚠️  Не удалось извлечь ID, пропускаю"
        return 1
    fi
    
    # Ищем дочерние документы
    child_files=$(find_child_documents "$file_id")
    
    if [ -n "$child_files" ]; then
        child_count=$(echo "$child_files" | wc -l | tr -d ' ')
        echo "    Найдено дочерних документов: $child_count"
        
        # Проверяем наличие раздела ДОЧЕРНИЕ ДОКУМЕНТЫ
        if grep -q "#### ДОЧЕРНИЕ ДОКУМЕНТЫ" "$file"; then
            echo "    ✅ Раздел ДОЧЕРНИЕ ДОКУМЕНТЫ существует"
            
            # Считаем существующие ссылки
            existing_links=$(grep -c "\[\[$file_id-" "$file" 2>/dev/null || echo "0")
            
            if [ "$existing_links" -ne "$child_count" ]; then
                echo "    ⚠️  Расхождение: ссылок $existing_links, дочерних $child_count"
                
                if [ "$fix_mode" = "true" ]; then
                    echo "    Исправляю..."
                    
                    # Создаем временный файл
                    temp_file="${file}.tmp"
                    
                    # Пересоздаем раздел с правильными ссылками
                    awk -v child_files="$child_files" '
                    BEGIN {
                        split(child_files, files, "\n")
                        in_children_section = 0
                        section_rebuilt = 0
                    }
                    
                    /#### ДОЧЕРНИЕ ДОКУМЕНТЫ/ {
                        in_children_section = 1
                        section_rebuilt = 1
                        print $0
                        # Добавляем все актуальные ссылки
                        for (i in files) {
                            if (files[i] != "") {
                                split(files[i], parts, "/")
                                child_filename = parts[length(parts)]
                                gsub(/\.md$/, "", child_filename)
                                # Проверяем, что это действительно дочерний документ
                                if (child_filename ~ "^" file_id "-") {
                                    print "- [[" child_filename "]]"
                                }
                            }
                        }
                        next
                    }
                    
                    in_children_section && /^[[:space:]]*Пока нет дочерних документов/ {
                        # Пропускаем эту строку
                        next
                    }
                    
                    in_children_section && (/^####/ || /^###/ || /^##/ || /^#/ || /^---/) {
                        in_children_section = 0
                    }
                    
                    {
                        if (!in_children_section) {
                            print $0
                        }
                    }
                    ' file_id="$file_id" "$file" > "$temp_file"
                    
                    mv "$temp_file" "$file"
                    echo "    ✅ Раздел исправлен"
                fi
            else
                echo "    ✅ Все ссылки корректны"
            fi
        else
            echo "    ⚠️  Раздел ДОЧЕРНИЕ ДОКУМЕНТЫ отсутствует"
            
            if [ "$fix_mode" = "true" ]; then
                echo "    Добавляю раздел..."
                
                # Добавляем раздел перед разделителем --- или в конец файла
                if grep -q "^---" "$file"; then
                    # Вставляем перед последним ---
                    awk -v child_files="$child_files" -v file_id="$file_id" '
                    BEGIN {
                        split(child_files, files, "\n")
                        children_added = 0
                        in_frontmatter = 0
                        frontmatter_end = 0
                    }
                    
                    /^---/ {
                        frontmatter_end++
                        if (frontmatter_end == 2 && !children_added) {
                            print $0
                            print ""
                            print "#### ДОЧЕРНИЕ ДОКУМЕНТЫ"
                            for (i in files) {
                                if (files[i] != "") {
                                    split(files[i], parts, "/")
                                    child_filename = parts[length(parts)]
                                    gsub(/\.md$/, "", child_filename)
                                    if (child_filename ~ "^" file_id "-") {
                                        print "- [[" child_filename "]]"
                                    }
                                }
                            }
                            children_added = 1
                            next
                        }
                    }
                    { print $0 }
                    ' "$file" > "${file}.tmp"
                else
                    # Добавляем в конец файла
                    echo "" >> "$file"
                    echo "#### ДОЧЕРНИЕ ДОКУМЕНТЫ" >> "$file"
                    for child in $child_files; do
                        child_name=$(basename "$child" .md)
                        echo "- [[$child_name]]" >> "$file"
                    done
                fi
                
                echo "    ✅ Раздел добавлен"
            fi
        fi
    else
        echo "    ℹ️  Нет дочерних документов"
        
        # Проверяем, есть ли раздел с "Пока нет дочерних документов"
        if ! grep -q "#### ДОЧЕРНИЕ ДОКУМЕНТЫ" "$file"; then
            echo "    ⚠️  Раздел ДОЧЕРНИЕ ДОКУМЕНТЫ отсутствует"
            
            if [ "$fix_mode" = "true" ]; then
                echo "    Добавляю раздел..."
                
                if grep -q "^---" "$file"; then
                    awk '
                    BEGIN {
                        children_added = 0
                        in_frontmatter = 0
                        frontmatter_end = 0
                    }
                    
                    /^---/ {
                        frontmatter_end++
                        if (frontmatter_end == 2 && !children_added) {
                            print $0
                            print ""
                            print "#### ДОЧЕРНИЕ ДОКУМЕНТЫ"
                            print "Пока нет дочерних документов."
                            children_added = 1
                            next
                        }
                    }
                    { print $0 }
                    ' "$file" > "${file}.tmp"
                else
                    echo "" >> "$file"
                    echo "#### ДОЧЕРНИЕ ДОКУМЕНТЫ" >> "$file"
                    echo "Пока нет дочерних документов." >> "$file"
                fi
                
                echo "    ✅ Раздел добавлен"
            fi
        fi
    fi
    
    echo ""
}
```

```typescript
import { App, Plugin, PluginSettingTab, Setting } from 'obsidian';
import { CreateMasterNoteCommand } from './src/commands/createMasterNote';
import { CreateChildNoteCommand } from './src/commands/createChildNote';

interface UCHSettings {
    defaultAuthor: string;
    autoUpdateReferences: boolean;
    hexPadding: number;
}

const DEFAULT_SETTINGS: UCHSettings = {
    defaultAuthor: 'system',
    autoUpdateReferences: true,
    hexPadding: 2
}

export default class UCHPlugin extends Plugin {
    settings: UCHSettings;

    async onload() {
        await this.loadSettings();
        
        // Регистрация команд
        this.addCommand({
            id: 'create-master-note',
            name: 'Create Master Document',
            callback: () => new CreateMasterNoteCommand(this).execute()
        });

        this.addCommand({
            id: 'create-child-note',
            name: 'Create Child Document',
            callback: () => new CreateChildNoteCommand(this).execute()
        });

        // Добавление вкладки настроек
        this.addSettingTab(new UCHSettingTab(this.app, this));
    }

    async loadSettings() {
        this.settings = Object.assign({}, DEFAULT_SETTINGS, await this.loadData());
    }

    async saveSettings() {
        await this.saveData(this.settings);
    }
}

class UCHSettingTab extends PluginSettingTab {
    plugin: UCHPlugin;

    constructor(app: App, plugin: UCHPlugin) {
        super(app, plugin);
        this.plugin = plugin;
    }

    display(): void {
        const {containerEl} = this;
        containerEl.empty();

        containerEl.createEl('h2', {text: 'UCH System Settings'});

        new Setting(containerEl)
            .setName('Default author')
            .setDesc('Author name for new documents')
            .addText(text => text
                .setPlaceholder('Enter your name')
                .setValue(this.plugin.settings.defaultAuthor)
                .onChange(async (value) => {
                    this.plugin.settings.defaultAuthor = value;
                    await this.plugin.saveSettings();
                }));
    }
}
```

```typescript
export class HexIdGenerator {
    private app: App;

    constructor(app: App) {
        this.app = app;
    }

    // Найти первый свободный мастер ID (уровень 1)
    async findFreeMasterId(): Promise<string> {
        const files = this.app.vault.getMarkdownFiles();
        const masterFiles = files.filter(file => 
            /^[0-9A-F]{2} - /.test(file.basename)
        );

        const ids = masterFiles.map(file => {
            const match = file.basename.match(/^([0-9A-F]{2})/);
            return match ? parseInt(match[1], 16) : 0;
        }).sort((a, b) => a - b);

        let expected = 0;
        for (const id of ids) {
            if (id > expected) break;
            expected = id + 1;
        }

        return expected.toString(16).toUpperCase().padStart(2, '0');
    }

    // Найти свободный дочерний ID
    async findFreeChildId(parentId: string): Promise<string> {
        const pattern = new RegExp(`^${parentId}-[0-9A-F]{2}`);
        const files = this.app.vault.getMarkdownFiles();
        
        const childFiles = files.filter(file => 
            pattern.test(file.basename)
        );

        const ids = childFiles.map(file => {
            const match = file.basename.match(new RegExp(`${parentId}-([0-9A-F]{2})`));
            return match ? parseInt(match[1], 16) : 0;
        }).sort((a, b) => a - b);

        let expected = 1;
        for (const id of ids) {
            if (id > expected) break;
            expected = id + 1;
        }

        return expected.toString(16).toUpperCase().padStart(2, '0');
    }
}
```

```typescript
interface UCHSettings {
    defaultAuthor: string;
    autoUpdateLinks: boolean;
    hexPadding: number;
}

// Значения по умолчанию
const DEFAULT_SETTINGS: UCHSettings = {
    defaultAuthor: 'system',
    autoUpdateLinks: true,
    hexPadding: 2
}

// Основной класс плагина
export default class UCHPlugin extends Plugin {
    settings: UCHSettings;

    // Загрузка плагина
    async onload() {
        console.log('UCH Plugin: Loading...');
        
        // Загрузка настроек
        await this.loadSettings();
        
        // 1. Добавляем команду в Command Palette
        this.addCommand({
            id: 'uch-hello-world',
            name: 'UCH: Say Hello',
            callback: () => {
                new Notice('👋 Hello from UCH Plugin!');
                console.log('UCH: Hello command executed');
            }
        });
        
        // 2. Добавляем команду создания тестового документа
        this.addCommand({
            id: 'uch-create-test-note',
            name: 'UCH: Create Test Note',
            callback: () => this.createTestNote()
        });
        
        // 3. Добавляем вкладку настроек
        this.addSettingTab(new UCHSettingTab(this.app, this));
        
        // 4. Добавляем иконку в левую панель
        this.addRibbonIcon('file-text', 'UCH Test', () => {
            new Notice('UCH Ribbon icon clicked!');
        });
        
        console.log('UCH Plugin: Loaded successfully!');
        new Notice('UCH Plugin loaded!');
    }
    
    // Выгрузка плагина
    onunload() {
        console.log('UCH Plugin: Unloading...');
    }
    
    // Создание тестовой заметки
    async createTestNote() {
        try {
            const fileName = `UCH-Test-${Date.now()}.md`;
            const content = `# UCH Test Note
            
This note was created by UCH Plugin!

**Timestamp:** ${new Date().toISOString()}

## Frontmatter Test
\`\`\`yaml
---
id: "TEST"
created: "${new Date().toISOString().split('T')[0]}"
author: "${this.settings.defaultAuthor}"
---
\`\`\`

## Next Steps
1. Implement HEX ID generation
2. Create master/child documents
3. Add automatic linking`;

            // Создаем файл в текущем vault
            const file = await this.app.vault.create(fileName, content);
            
            new Notice(`✅ Created: ${fileName}`);
            console.log('UCH: Created test note:', fileName);
            
            // Открываем созданный файл
            await this.app.workspace.getLeaf().openFile(file);
            
        } catch (error) {
            new Notice(`❌ Error: ${error.message}`);
            console.error('UCH: Error creating note:', error);
        }
    }
    
    // Загрузка настроек
    async loadSettings() {
        this.settings = Object.assign({}, DEFAULT_SETTINGS, await this.loadData());
    }
    
    // Сохранение настроек
    async saveSettings() {
        await this.saveData(this.settings);
    }
}

// Класс для вкладки настроек
class UCHSettingTab extends PluginSettingTab {
    plugin: UCHPlugin;
    
    constructor(app: App, plugin: UCHPlugin) {
        super(app, plugin);
        this.plugin = plugin;
    }
    
    display(): void {
        const { containerEl } = this;
        containerEl.empty();
        
        // Заголовок
        containerEl.createEl('h2', { text: 'UCH System Settings' });
        
        // Настройка автора
        new Setting(containerEl)
            .setName('Default author')
            .setDesc('Author name for new documents')
            .addText(text => text
                .setPlaceholder('Enter your name')
                .setValue(this.plugin.settings.defaultAuthor)
                .onChange(async (value) => {
                    this.plugin.settings.defaultAuthor = value;
                    await this.plugin.saveSettings();
                }));
        
        // Настройка автообновления ссылок
        new Setting(containerEl)
            .setName('Auto-update links')
            .setDesc('Automatically update parent/child links')
            .addToggle(toggle => toggle
                .setValue(this.plugin.settings.autoUpdateLinks)
                .onChange(async (value) => {
                    this.plugin.settings.autoUpdateLinks = value;
                    await this.plugin.saveSettings();
                }));
        
        // Настройка padding для HEX
        new Setting(containerEl)
            .setName('HEX padding')
            .setDesc('Number of characters for HEX IDs (2 = 00, 01, ...)')
            .addSlider(slider => slider
                .setLimits(2, 4, 1)
                .setValue(this.plugin.settings.hexPadding)
                .setDynamicTooltip()
                .onChange(async (value) => {
                    this.plugin.settings.hexPadding = value;
                    await this.plugin.saveSettings();
                }));
    }
}
```
