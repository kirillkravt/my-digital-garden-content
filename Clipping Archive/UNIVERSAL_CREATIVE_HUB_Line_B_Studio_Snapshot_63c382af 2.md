## 🔧 ПРАКТИЧЕСКАЯ РЕАЛИЗАЦИЯ

### 1. **ГЕНЕРАЦИЯ ID**
```typescript
class DocumentNumbering {
  private static counters: Map<string, number> = new Map();
  
  static generateId(prefix: string = '00'): string {
    const level = prefix.split('-').length;
    const baseKey = prefix.substring(0, prefix.lastIndexOf('-') || 0);
    
    if (!this.counters.has(baseKey)) {
      this.counters.set(baseKey, 0);
    }
    
    let counter = this.counters.get(baseKey)! + 1;
    this.counters.set(baseKey, counter);
    
    // Hex форматирование в зависимости от уровня
    const hexLength = level * 2; // 2, 4, 6, 8...
    const hexCounter = counter.toString(16).padStart(2, '0');
    
    return level === 1 
      ? hexCounter.padStart(2, '0')
      : `${prefix}-${hexCounter}`;
  }
}
```

# 🗄️ ПОЗИЦИОННАЯ HEX-СИСТЕМА ДЛЯ БАЗЫ ДАННЫХ ЗАМЕТОК

## ✅ **ДА, ИДЕАЛЬНО ПОДХОДИТ ДЛЯ БАЗЫ ДАННЫХ**

Позиционная hex-система с 2 символами на уровень — **идеальная** схема нумерации для управления большими массивами документов в БД. Вот почему:

## 🏗️ **АРХИТЕКТУРНЫЕ ПРЕИМУЩЕСТВА ДЛЯ БД**

### 1. **ЕСТЕСТВЕННЫЕ ПЕРВИЧНЫЕ КЛЮЧИ**
```sql
CREATE TABLE notes (
  id VARCHAR(32) PRIMARY KEY,      -- '00-02-01-03-01'
  parent_id VARCHAR(32),           -- '00-02-01-03' (легко вычислить)
  title VARCHAR(255),
  content TEXT,
  created_at TIMESTAMP,
  level TINYINT,                   -- 1, 2, 3, 4, 5...
  FOREIGN KEY (parent_id) REFERENCES notes(id)
);
```

**Индексы работают идеально:**
- `id` - PRIMARY KEY (сортировка по иерархии)
- `parent_id` - FOREIGN KEY + INDEX (быстрый доступ к детям)
- `level` - INDEX (фильтрация по уровню)

### 2. **ЭФФЕКТИВНЫЕ ЗАПРОСЫ**

```sql
-- Найти всех детей документа (прямые потомки)
SELECT * FROM notes 
WHERE parent_id = '00-02-01-03';

-- Найти все документы уровня 3
SELECT * FROM notes 
WHERE level = 3;

-- Найти поддерево (все потомки)
SELECT * FROM notes 
WHERE id LIKE '00-02-01-%';

-- Получить путь к документу (все предки)
WITH RECURSIVE ancestors AS (
  SELECT id, parent_id, title, level
  FROM notes WHERE id = '00-02-01-03-01'
  UNION ALL
  SELECT n.id, n.parent_id, n.title, n.level
  FROM notes n
  JOIN ancestors a ON n.id = a.parent_id
)
SELECT * FROM ancestors ORDER BY level ASC;
```

### 3. **ВЫЧИСЛЕНИЯ БЕЗ JOIN**
```sql
-- Получить родителя без JOIN (O(1) операция)
SELECT SUBSTRING_INDEX(id, '-', LENGTH(id) - LENGTH(REPLACE(id, '-', ''))) 
FROM notes WHERE id = '00-02-01-03-01';
-- Результат: '00-02-01-03'

-- Получить уровень из ID
SELECT (LENGTH(id) - LENGTH(REPLACE(id, '-', ''))) / 2 + 1 AS level
FROM notes WHERE id = '00-02-01-03';
-- Результат: 4
```

## 📊 **ОБЪЁМЫ И МАСШИБАБИРУЕМОСТЬ**

### **ЁМКОСТЬ КАЖДОГО УРОВНЯ:**
- **2 hex-символа = 256 вариантов (00-FF)**
- **На каждом уровне:** 255 возможных детей (00 зарезервирован для корня)

```typescript
class NoteIDManager {
  // Генерация нового ID для уровня
  static generateChildId(parentId: string): string {
    if (!parentId) return this.generateRootId();
    
    const lastId = await this.getLastChildId(parentId);
    const nextHex = this.incrementHex(lastId?.split('-').pop() || '00');
    
    return `${parentId}-${nextHex}`;
  }
  
  // Резервирование диапазона ID
  static reserveRange(parentId: string, count: number): string[] {
    const ids: string[] = [];
    for (let i = 0; i < count; i++) {
      ids.push(`${parentId}-${(i + 1).toString(16).padStart(2, '0')}`);
    }
    return ids;
  }
}

// Глобальный сервис для управления AudioContext
class AudioActivator {
  private static instance: AudioActivator;
  private audioContext: AudioContext | null = null;
  
  static getInstance() {
    if (!AudioActivator.instance) {
      AudioActivator.instance = new AudioActivator();
    }
    return AudioActivator.instance;
  }
  
  async activate() {
    if (!this.audioContext) {
      this.audioContext = new AudioContext();
    }
    
    if (this.audioContext.state === 'suspended') {
      await this.audioContext.resume();
      return true;
    }
    return false;
  }
  
  getContext() {
    return this.audioContext;
  }
}
```

#!/bin/bash
# Скрипт для генерации новых ID заметок в системе hex-нумерации - ФИНАЛЬНАЯ ВЕРСИЯ

set -e  # Выход при ошибке

# Цвета для вывода
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# Функция для вывода цветного текста
print_color() {
    echo -e "${1}${2}${NC}"
}

# Функция для расчета уровня по ID
calculate_level() {
    local id="$1"
    if [ -z "$id" ]; then
        echo 0
    else
        # Количество дефисов определяет уровень минус 1
        local num_dashes=$(echo "$id" | tr -cd '-' | wc -c)
        echo $((num_dashes + 1))
    fi
}

# Проверка наличия родительского ID
if [ -z "$1" ]; then
    print_color $YELLOW "Использование: $0 <parent_id> [note_title]"
    print_color $BLUE "Примеры:"
    print_color $BLUE "  $0 00-02-01 \"Новая задача\""
    print_color $BLUE "  $0 00-02 \"Новый компонент\""
    print_color $BLUE "  $0 \"\" \"Новый проект\" (для корневого уровня)"
    exit 1
fi

PARENT_ID="$1"
NOTE_TITLE="$2"
VAULT_PATH="/Users/kirillkravcov/obsidian/my-digital-garden-content/uch-docs"

# Переходим в директорию vault
cd "$VAULT_PATH" || {
    print_color $RED "Ошибка: Не могу перейти в $VAULT_PATH"
    exit 1
}

print_color $GREEN "🎯 Генерация нового ID для системы нумерации UCH"
print_color $BLUE "=============================================="

# Если пустой parent_id - генерируем корневой ID
if [ -z "$PARENT_ID" ]; then
    print_color $YELLOW "Генерация ID для корневого уровня..."
    
    # Ищем максимальный существующий корневой ID
    MAX_ROOT=$(find . -maxdepth 1 -name "*.md" -type f \
        | grep -E '^\./[0-9a-fA-F]{2} - .*\.md$' \
        | sed 's|^\./\([0-9a-fA-F]\{2\}\) - .*\.md$|\1|' \
        | sort -r | head -n1)
    
    if [ -z "$MAX_ROOT" ]; then
        NEW_ID="00"
    else
        # Конвертируем hex в decimal, увеличиваем, обратно в hex
        DEC_VAL=$((16#$MAX_ROOT))
        NEXT_VAL=$((DEC_VAL + 1))
        NEW_ID=$(printf "%02x" $NEXT_VAL)
    fi
    
    PARENT_LEVEL=0
    NEW_LEVEL=1
else
    # Проверяем формат parent_id
    if [[ ! "$PARENT_ID" =~ ^[0-9a-fA-F]{2}(-[0-9a-fA-F]{2})*$ ]]; then
        print_color $RED "Ошибка: Неверный формат parent_id: $PARENT_ID"
        print_color $YELLOW "Формат должен быть: XX-XX-XX (hex, разделитель -)"
        exit 1
    fi
    
    # Определяем уровень родительского документа
    PARENT_LEVEL=$(calculate_level "$PARENT_ID")
    NEW_LEVEL=$((PARENT_LEVEL + 1))
    
    # Ищем существующие документы с этим parent_id
    print_color $BLUE "Поиск существующих детей для $PARENT_ID..."
    
    # Паттерн для поиска детей: parent_id-XX
    CHILDREN_PATTERN="${PARENT_ID}-[0-9a-fA-F]\{2\}"
    
    # Ищем максимальный child ID
    MAX_CHILD=$(find . -name "*.md" -type f \
        | grep -E "${CHILDREN_PATTERN} - " \
        | sed "s|.*/${PARENT_ID}-\([0-9a-fA-F]\{2\}\) - .*\.md$|\1|" \
        | sort -r | head -n1)
    
    if [ -z "$MAX_CHILD" ]; then
        NEW_CHILD_ID="01"
    else
        # Увеличиваем hex значение
        DEC_VAL=$((16#$MAX_CHILD))
        NEXT_VAL=$((DEC_VAL + 1))
        NEW_CHILD_ID=$(printf "%02x" $NEXT_VAL)
    fi
    
    NEW_ID="${PARENT_ID}-${NEW_CHILD_ID}"
fi

print_color $BLUE "  Родительский уровень: $PARENT_LEVEL"
print_color $BLUE "  Новый уровень: $NEW_LEVEL"

# Генерируем имя файла
if [ -n "$NOTE_TITLE" ]; then
    FILENAME="${NEW_ID} - ${NOTE_TITLE}.md"
    
    # Запрашиваем теги
    print_color $YELLOW "Введите теги через пробел (например: @bug @audio @high-priority):"
    read -r TAGS_INPUT
    
    # Создаем файл с базовой структурой
    cat > "$FILENAME" << FILE_CONTENT
# ${NEW_ID} - ${NOTE_TITLE}

## 📋 ОБЩАЯ ИНФОРМАЦИЯ
- **Полный путь**: Уровень ${NEW_LEVEL}
- **Статус**: 🆕 НОВЫЙ ДОКУМЕНТ
- **Родитель**: ${PARENT_ID:-[корень]}
- **ID**: \`${NEW_ID}\`
- **Уровень**: ${NEW_LEVEL}
- **Теги**: ${TAGS_INPUT:-@todo}

## 🎯 ОПИСАНИЕ

*Добавьте описание здесь...*

## 📋 ЗАДАЧИ

- [ ] Задача 1
- [ ] Задача 2

## 🔗 СВЯЗАННЫЕ ДОКУМЕНТЫ

### РОДИТЕЛЬСКИЕ:

### СМЕЖНЫЕ:
*Добавьте ссылки...*

---
*Создано: $(date +%Y-%m-%d)*
*Статус: Черновик*
*Уровень: ${NEW_LEVEL}*
FILE_CONTENT
    
    print_color $GREEN "✅ Создан файл: $FILENAME"
    
    # Если есть parent_id, обновляем родительский файл
    if [ -n "$PARENT_ID" ] && [ "$PARENT_ID" != "00" ]; then
        PARENT_FILE=$(find . -name "${PARENT_ID} - *.md" -type f | head -n1)
        
        if [ -n "$PARENT_FILE" ]; then
            print_color $BLUE "📝 Обновляю родительский файл: $PARENT_FILE"
            
            # Проверяем есть ли уже ссылка на этот документ
            if ! grep -q "\[\[${NEW_ID} -" "$PARENT_FILE"; then
                # Добавляем в раздел дочерних документов
                if grep -q "### ДОЧЕРНИЕ:" "$PARENT_FILE"; then
                    # Вставляем после "### ДОЧЕРНИЕ:"
                    sed -i '' "/### ДОЧЕРНИЕ:/a\\
- [[${NEW_ID} - ${NOTE_TITLE}]]" "$PARENT_FILE"
                else
                    # Добавляем раздел
                    echo "" >> "$PARENT_FILE"
                    echo "### ДОЧЕРНИЕ:" >> "$PARENT_FILE"
                    echo "- [[${NEW_ID} - ${NOTE_TITLE}]]" >> "$PARENT_FILE"
                fi
                print_color $GREEN "✅ Ссылка добавлена в родительский документ"
            fi
        else
            print_color $YELLOW "⚠️ Родительский файл не найден: $PARENT_ID"
        fi
    fi
else
    # Только вывод ID без создания файла
    print_color $GREEN "📋 Результат генерации:"
    print_color $BLUE "  Новый ID: ${NEW_ID}"
    print_color $BLUE "  Уровень: ${NEW_LEVEL}"
    print_color $BLUE "  Родитель: ${PARENT_ID:-[корень]}"
    print_color $BLUE "  Родительский уровень: ${PARENT_LEVEL}"
    print_color $YELLOW ""
    print_color $YELLOW "Чтобы создать файл, укажите название:"
    print_color $YELLOW "  $0 \"$PARENT_ID\" \"Название заметки\""
fi

print_color $BLUE "=============================================="
print_color $GREEN "🎉 Готово! Система hex-нумерации активна."
EOF
