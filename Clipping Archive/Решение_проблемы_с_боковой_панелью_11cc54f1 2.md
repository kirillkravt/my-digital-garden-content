## 🔍 **Анализ проблемы**

Частые причины отсутствия боковой панели:

1. **Неправильные пути к CSS/JS файлам**
2. **Ошибки в шаблонах Bootstrap 5** (классы, структура сетки)
3. **Отсутствие контекста для боковой панели** в Django view
4. **Проблемы с медиа-файлами или статикой**

## 🛠️ **Пошаговое решение**

### 3. **Обновим view для главной страницы**

```python
# blog/views.py
from django.shortcuts import render
from .models import Category, Article, Tag
from django.db.models import Count

def home_view(request):
    # Последние статьи
    latest_articles = Article.objects.filter(
        status='published'
    ).order_by('-created_at')[:5]
    
    # Все категории
    categories = Category.objects.all()
    
    # Популярные теги
    popular_tags = Tag.objects.annotate(
        num_articles=Count('article')
    ).order_by('-num_articles')[:10]
    
    context = {
        'latest_articles': latest_articles,
        'categories': categories,
        'popular_tags': popular_tags,
        'featured_articles': latest_articles[:3],
    }
    
    return render(request, 'blog/home.html', context)
```

from .models import Category, Tag
from django.db.models import Count

def sidebar_data(request):
    """Добавляет данные для боковой панели во все шаблоны"""
    categories = Category.objects.all()
    popular_tags = Tag.objects.annotate(
        num_articles=Count('article')
    ).order_by('-num_articles')[:10]
    
    return {
        'categories': categories,
        'popular_tags': popular_tags,
    }

# blog/views.py (продолжение)
from django.shortcuts import render, get_object_or_404
from django.core.paginator import Paginator

def article_list(request):
    """Список всех статей"""
    articles_list = Article.objects.filter(status='published').order_by('-created_at')
    
    # Пагинация
    paginator = Paginator(articles_list, 10)
    page_number = request.GET.get('page')
    page_obj = paginator.get_page(page_number)
    
    return render(request, 'blog/article_list.html', {
        'page_obj': page_obj,
        'articles': page_obj.object_list,
    })

def article_detail(request, slug):
    """Детальная страница статьи"""
    article = get_object_or_404(Article, slug=slug, status='published')
    
    # Увеличиваем просмотры
    article.views = (article.views or 0) + 1
    article.save(update_fields=['views'])
    
    # Связанные статьи
    related_articles = Article.objects.filter(
        status='published',
        category=article.category
    ).exclude(id=article.id)[:3]
    
    return render(request, 'blog/article_detail.html', {
        'article': article,
        'related_articles': related_articles,
    })

def category_list(request):
    """Список всех категорий"""
    categories = Category.objects.annotate(
        article_count=Count('article', filter=Q(article__status='published'))
    ).filter(article_count__gt=0)
    
    return render(request, 'blog/category_list.html', {
        'categories': categories,
    })

def category_detail(request, slug):
    """Статьи в категории"""
    category = get_object_or_404(Category, slug=slug)
    articles = Article.objects.filter(
        category=category,
        status='published'
    ).order_by('-created_at')
    
    return render(request, 'blog/category_detail.html', {
        'category': category,
        'articles': articles,
    })

# blog/context_processors.py
from django.db.models import Count, Q
from blog.models import Category, Article, Tag

def sidebar_data(request):
    """
    Добавляет данные для боковой панели во все шаблоны
    """
    try:
        # Аннотируем категории с количеством статей
        categories = Category.objects.annotate(
            article_count=Count('article', filter=Q(article__status='published'))
        ).filter(article_count__gt=0)
        
        # Популярные теги
        popular_tags = Tag.objects.annotate(
            article_count=Count('article', filter=Q(article__status='published'))
        ).filter(article_count__gt=0).order_by('-article_count')[:10]
        
        # Последние статьи
        latest_articles = Article.objects.filter(
            status='published'
        ).order_by('-created_at')[:5]
        
        return {
            'categories': categories,
            'popular_tags': popular_tags,
            'latest_articles': latest_articles,
        }
    except:
        # Если база данных ещё не готова, возвращаем пустые данные
        return {
            'categories': [],
            'popular_tags': [],
            'latest_articles': [],
        }

# uch/apps/blog/context_processors.py
from django.db.models import Count, Q

def sidebar_data(request):
    """Данные для боковой панели"""
    try:
        from .models import Category, Article
        from taggit.models import Tag
        
        # Категории с количеством статей
        categories = Category.objects.annotate(
            article_count=Count('article', filter=Q(article__status='published'))
        ).filter(is_active=True, article_count__gt=0)
        
        # Популярные теги
        popular_tags = Tag.objects.annotate(
            num_times=Count('taggit_taggeditem_items')
        ).order_by('-num_times')[:15]
        
        # Последние статьи
        latest_articles = Article.objects.filter(
            status='published'
        ).order_by('-created_at')[:5]
        
        return {
            'categories': categories,
            'popular_tags': popular_tags,
            'latest_articles': latest_articles,
        }
        
    except Exception as e:
        # Если база данных ещё не готова
        print(f"⚠️  Context processor error: {e}")
        return {
            'categories': [],
            'popular_tags': [],
            'latest_articles': [],
        }


def blog_categories(request):
    """Категории блога для навигации"""
    try:
        from .models import Category
        categories = Category.objects.filter(is_active=True)[:10]
        return {'blog_categories': categories}
    except:
        return {'blog_categories': []}


def blog_stats(request):
    """Статистика блога"""
    try:
        from .models import Article, Category
        return {
            'total_articles': Article.objects.filter(status='published').count(),
            'total_categories': Category.objects.filter(is_active=True).count(),
        }
    except:
        return {
            'total_articles': 0,
            'total_categories': 0,
        }
EOF
