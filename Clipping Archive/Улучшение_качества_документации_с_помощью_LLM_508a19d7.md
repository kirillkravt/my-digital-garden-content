### Текущая реализация (предполагаемая структура):

```python
# Текущий проблемный код
class DocsGenerator:
    def __init__(self, llm_service_url="http://localhost:8003"):
        self.llm_service_url = llm_service_url
    
    async def generate_docstring(self, file_content, language="python"):
        """Генерирует docstring для файла"""
        prompt = f"""
        Сгенерируй документацию для следующего кода:
        
        {file_content}
        
        Верни только docstring.
        """
        return await self._call_llm(prompt)
    
    async def analyze_commit_messages(self, commit_messages):
        """Анализирует коммит сообщения"""
        prompt = f"""
        Проанализируй коммит сообщения: {commit_messages}
        
        Что изменилось?
        """
        return await self._call_llm(prompt)
```

### Проблемы текущих промптов:

1. **❌ Отсутствие контекста изменений**
   - Не передаётся diff, только полный файл
   - Нет информации о том, что именно было изменено

2. **❌ Нечёткие инструкции**
   - "Верни только docstring" - но в каком формате?
   - Нет примеров хороших docstrings

3. **❌ Отсутствие структуры ответа**
   - LLM может вернуть что угодно в любом формате
   - Нет валидации формата

4. **❌ Нет обработки edge cases**
   - Что если файл слишком большой?
   - Что если код не документируемый?

5. **❌ Отсутствие системы оценки качества**
   - Не проверяем, действительно ли сгенерирована документация

    ```

    ```{language}
    {file_content}
    ```

    1. Проанализируй diff и пойми, что именно изменилось

    ```python
    def calculate_metrics(data: List[Dict], threshold: float = 0.5) -> Dict[str, float]:
        \"\"\"Рассчитывает метрики качества на основе входных данных.
        
        Args:
            data: Список словарей с полями 'actual' и 'predicted'
            threshold: Порог для бинаризации предсказаний (default: 0.5)
        
        Returns:
            Dict с ключами: 'accuracy', 'precision', 'recall', 'f1_score'
        
        Raises:
            ValueError: Если data пуст или отсутствуют обязательные поля

            >>> data = [{'actual': 1, 'predicted': 0.8}, {'actual': 0, 'predicted': 0.3}]
            >>> calculate_metrics(data)
            {'accuracy': 0.5, 'precision': 0.5, 'recall': 0.5, 'f1_score': 0.5}
        \"\"\"
    ```
    
    ## ТРЕБОВАНИЯ К ОТВЕТУ:
    1. Документируй ТОЛЬКО изменённые или новые функции/классы
    2. Используй Google style docstrings для Python (или соответствующий стиль для {language})
    3. Обязательно включай:
       - Описание назначения
       - Параметры (Args:)
       - Возвращаемое значение (Returns:)
       - Исключения (Raises:) если есть
       - Пример использования (Example:)

    ## ФОРМАТ ОТВЕТА (строго JSON):

        "file_path": "{file_path}",
        "language": "{language}",

### 1. Система оценки качества генерации:

```python
async def evaluate_doc_quality(self, generated_doc, original_code):
    """Оценивает качество сгенерированной документации"""
    
    metrics = {
        "has_description": bool(re.search(r'"""\s*\w+', generated_doc)),
        "has_args": "Args:" in generated_doc if ":param" not in generated_doc else True,
        "has_returns": "Returns:" in generated_doc or ":return:" in generated_doc,
        "has_examples": "Example:" in generated_doc or ">>>" in generated_doc,
        "has_raises": "Raises:" in generated_doc or ":raises:" in generated_doc,
        "length_reasonable": 50 < len(generated_doc) < 2000
    }
    
    quality_score = sum(metrics.values()) / len(metrics)
    
    return {
        "score": quality_score,
        "metrics": metrics,
        "needs_regeneration": quality_score < 0.6,
        "missing_elements": [k for k, v in metrics.items() if not v]
    }
```

### 2. Обработка ошибок и ретраи:

```python
async def _call_llm_with_retry(self, prompt, max_retries=3):
    """Вызов LLM с ретраями и экспоненциальной задержкой"""
    
    for attempt in range(max_retries):
        try:
            response = await self._call_llm(prompt)
            
            # Проверяем, что ответ не пустой
            if not response or len(response.strip()) < 10:
                raise ValueError("Empty or too short response")
            
            # Пытаемся распарсить JSON если ожидаем JSON
            if prompt.strip().endswith('ВЕРНИ ТОЛЬКО JSON'):
                try:
                    json.loads(response)
                except json.JSONDecodeError:
                    # Пробуем извлечь JSON из текста
                    json_match = re.search(r'\{.*\}', response, re.DOTALL)
                    if json_match:
                        response = json_match.group()
                        json.loads(response)  # Проверяем валидность
                    else:
                        raise ValueError("Response is not valid JSON")
            
            return response
            
        except Exception as e:
            if attempt == max_retries - 1:
                # Логируем ошибку для мониторинга
                await self._log_generation_error(prompt, str(e))
                raise
            
            wait_time = 2 ** attempt  # Экспоненциальная задержка
            await asyncio.sleep(wait_time)
```

async def _log_generation_error(self, prompt, error):
    """Логирует ошибки генерации для мониторинга"""
    
    log_entry = {
        "timestamp": datetime.utcnow().isoformat(),
        "error_type": type(error).__name__,
        "error_message": str(error),
        "prompt_preview": prompt[:500] + "...",
        "prompt_length": len(prompt),
        "service": "docs_generator"
    }
    
    # Отправляем в систему мониторинга
    await self._send_to_monitoring(log_entry)
```

```python
class DocsGenerator:
    def __init__(self, llm_url: str = "http://localhost:8003"):
        self.llm_url = llm_url
        self.default_model = "balanced"
        self.metrics_collector = MetricsCollector()  # Для сбора метрик
        
    async def _call_llm_with_retry(self, prompt: str, max_retries: int = 3) -> Optional[str]:
        """Вызов LLM с ретраями и сбором метрик"""
        
        for attempt in range(max_retries):
            try:
                start_time = time.time()
                
                response = await self._call_llm(prompt)
                
                # Собираем метрики
                await self.metrics_collector.collect(
                    prompt_length=len(prompt),
                    response_length=len(response) if response else 0,
                    duration=time.time() - start_time,
                    success=True,
                    attempt=attempt + 1
                )
                
                # Валидация JSON если нужно
                if "ВЕРНИ ТОЛЬКО JSON" in prompt:
                    try:
                        json.loads(response)
                    except json.JSONDecodeError:
                        # Пытаемся извлечь JSON
                        json_match = re.search(r'\{.*\}', response, re.DOTALL)
                        if json_match:
                            response = json_match.group()
                            json.loads(response)  # Проверяем валидность
                        else:
                            raise ValueError("Invalid JSON response")
                
                return response
                
            except Exception as e:
                await self.metrics_collector.collect(
                    success=False,
                    error=str(e),
                    attempt=attempt + 1
                )
                
                if attempt == max_retries - 1:
                    logger.error(f"LLM call failed after {max_retries} attempts: {e}")
                    return None
                    
                wait_time = 2 ** attempt  # Экспоненциальная задержка
                await asyncio.sleep(wait_time)
    
    async def _call_llm(self, prompt: str) -> str:
        """Реальный вызов LLM сервиса"""
        async with aiohttp.ClientSession() as session:
            payload = {
                "model": self.default_model,
                "prompt": prompt,
                "max_tokens": 2000,
                "temperature": 0.3,  # Низкая температура для более детерминированных ответов
                "stream": False
            }
            
            async with session.post(f"{self.llm_url}/generate", json=payload) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    return data.get("response", "")
                else:
                    raise Exception(f"LLM service returned {resp.status}")
```

class MetricsCollector:
    """Сборщик метрик вызовов LLM."""
    
    def __init__(self):
        self.calls = []
    
    async def collect(self, **kwargs):
        self.calls.append(kwargs)
    
    def get_stats(self) -> dict:
        if not self.calls:
            return {"total": 0, "success_rate": 0}
        
        total = len(self.calls)
        successful = sum(1 for c in self.calls if c.get("success", False))
        return {
            "total": total,
            "success_rate": successful / total if total > 0 else 0,
            "avg_duration": sum(c.get("duration", 0) for c in self.calls) / total if total > 0 else 0
        }


class DocsGenerator:
    """Генератор документации с использованием LLM сервиса."""
    
    def __init__(self, llm_url: str = "http://localhost:8003"):
        """Инициализация генератора документации."""
        self.llm_url = llm_url
        self.default_model = "balanced"
        self.timeout = aiohttp.ClientTimeout(total=60, connect=10, sock_read=30)
        self.metrics_collector = MetricsCollector()
        logger.info(f"DocsGenerator initialized with LLM URL: {llm_url}")
    
    async def _call_llm(self, prompt: str, max_tokens: int = 2000, temperature: float = 0.3) -> Optional[str]:
        """Вызов LLM сервиса."""
        async with aiohttp.ClientSession(timeout=self.timeout) as session:
            async with session.post(
                f"{self.llm_url}/generate",
                json={
                    "model": self.default_model,
                    "prompt": prompt,
                    "max_tokens": max_tokens,
                    "temperature": temperature,
                    "stream": False
                }
            ) as response:
                if response.status == 200:
                    result = await response.json()
                    return result.get("response", "")
                else:
                    error_text = await response.text()
                    raise Exception(f"LLM service returned {response.status}: {error_text}")
    
    async def _call_llm_with_retry(self, prompt: str, max_retries: int = 3, 
                                    max_tokens: int = 2000, temperature: float = 0.3) -> Optional[str]:
        """Вызов LLM с ретраями и сбором метрик."""
        for attempt in range(max_retries):
            try:
                start_time = time.time()
                response = await self._call_llm(prompt, max_tokens, temperature)
                
                await self.metrics_collector.collect(
                    prompt_length=len(prompt),
                    response_length=len(response) if response else 0,
                    duration=time.time() - start_time,
                    success=True,
                    attempt=attempt + 1
                )
                
                return response
                
            except Exception as e:
                await self.metrics_collector.collect(
                    success=False,
                    error=str(e),
                    attempt=attempt + 1
                )
                
                if attempt == max_retries - 1:
                    logger.error(f"LLM call failed after {max_retries} attempts: {e}")
                    return None
                
                wait_time = 2 ** attempt
                logger.warning(f"LLM call failed, retrying in {wait_time}s (attempt {attempt + 1}/{max_retries})")
                await asyncio.sleep(wait_time)
        
        return None
    
    def _extract_json(self, text: str) -> Optional[dict]:
        """Извлечение JSON из текста."""
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            json_match = re.search(r'\{.*\}', text, re.DOTALL)
            if json_match:
                try:
                    return json.loads(json_match.group())
                except json.JSONDecodeError:
                    pass
        return None

@click.group()
@click.option('--llm-url', default='http://localhost:8003', help='URL LLM сервиса')
@click.pass_context
def cli(ctx, llm_url):
    """Генератор документации с использованием LLM."""
    ctx.ensure_object(dict)
    ctx.obj['generator'] = DocsGenerator(llm_url)


@cli.command()
@click.argument('code', required=False)
@click.option('--file', '-f', type=click.Path(exists=True), help='Путь к файлу с кодом')
@click.option('--language', '-l', default='python', help='Язык программирования')
@click.option('--diff', '-d', help='Git diff контекст')
@click.option('--commits', '-c', multiple=True, help='Сообщения коммитов')
@click.option('--filepath', help='Путь к файлу')
@click.pass_context
def docstring(ctx, code, file, language, diff, commits, filepath):
    """Генерировать docstring для кода."""
    generator: DocsGenerator = ctx.obj['generator']
    
    if file:
        with open(file, 'r') as f:
            code = f.read()
        if not filepath:
            filepath = file
    elif not code:
        click.echo("Укажите код через аргумент или файл через --file")
        sys.exit(1)
    
    result = asyncio.run(generator.generate_docstring(
        code, diff, list(commits), filepath, language
    ))
    
    if result:
        click.echo(result)
    else:
        click.echo("Ошибка генерации docstring", err=True)
        sys.exit(1)


@cli.command()
@click.argument('repo_name')
@click.option('--files', '-f', multiple=True, help='Список файлов в формате path:description')
@click.option('--readme', '-r', type=click.Path(exists=True), help='Путь к текущему README')
@click.option('--change-type', default='update', help='Тип изменений')
@click.pass_context
def readme(ctx, repo_name, files, readme, change_type):
    """Генерировать секцию для README."""
    generator: DocsGenerator = ctx.obj['generator']
    
    current_readme = None
    if readme:
        with open(readme, 'r') as f:
            current_readme = f.read()
    
    files_summary = []
    for f in files:
        if ':' in f:
            path, desc = f.split(':', 1)
            files_summary.append({'path': path, 'description': desc})
        else:
            files_summary.append({'path': f, 'description': 'Новый файл'})
    
    result = asyncio.run(generator.generate_readme_section(
        files_summary, repo_name, current_readme, change_type
    ))
    
    if result:
        click.echo(result)
    else:
        click.echo("Ошибка генерации README секции", err=True)
        sys.exit(1)

        """Расчет статистик."""

            'stdev': statistics.stdev(values) if len(values) > 1 else 0,
