```bash
# Создать минимальный рабочий views.py
cat > uch/apps/blog/views.py << 'EOF'
from django.shortcuts import render, get_object_or_404
from django.views.generic import ListView, DetailView
from django.db.models import Count, Q
from .models import Article, Category

# Context processor для боковой панели
def sidebar_context(request):
    """Контекст для боковой панели (категории и последние статьи)"""
    # Категории
    categories = Category.objects.all()
    
    # Последние опубликованные статьи
    recent_articles = Article.objects.filter(
        status='published'
    ).order_by('-created_at')[:5]
    
    return {
        'blog_categories': categories,
        'blog_recent_articles': recent_articles,
    }

# Главная страница
class HomeView(ListView):
    template_name = 'blog/home.html'
    model = Article
    context_object_name = 'articles'
    
    def get_queryset(self):
        return Article.objects.filter(status='published').order_by('-created_at')
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context.update(sidebar_context(self.request))
        return context

# Список всех статей
class ArticleListView(ListView):
    template_name = 'blog/article_list.html'
    model = Article
    context_object_name = 'articles'
    
    def get_queryset(self):
        queryset = Article.objects.filter(status='published').order_by('-created_at')
        
        # Поиск
        search_query = self.request.GET.get('search', '')
        if search_query:
            queryset = queryset.filter(
                Q(title__icontains=search_query) | 
                Q(content__icontains=search_query)
            )
        
        return queryset
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context.update(sidebar_context(self.request))
        context['search_query'] = self.request.GET.get('search', '')
        return context

# Детальная страница статьи
class ArticleDetailView(DetailView):
    template_name = 'blog/article_detail.html'
    model = Article
    context_object_name = 'article'
    
    def get_queryset(self):
        return Article.objects.filter(status='published')
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context.update(sidebar_context(self.request))
        return context

# Список категорий
class CategoryListView(ListView):
    template_name = 'blog/category_list.html'
    model = Category
    context_object_name = 'categories'
    
    def get_queryset(self):
        return Category.objects.all().order_by('name')
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context.update(sidebar_context(self.request))
        return context

# Статьи в категории
class CategoryDetailView(ListView):
    template_name = 'blog/category_detail.html'
    model = Article
    context_object_name = 'articles'
    
    def get_queryset(self):
        category_slug = self.kwargs['slug']
        self.category = get_object_or_404(Category, slug=category_slug)
        return Article.objects.filter(
            category=self.category,
            status='published'
        ).order_by('-created_at')
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context.update(sidebar_context(self.request))
        context['category'] = self.category
        return context
EOF
```

        if search_query:
            queryset = queryset.filter(
                Q(title__icontains=search_query) | 
                Q(content__icontains=search_query)
            )
        
        # Фильтрация по тегу
        tag = self.request.GET.get('tag', '')
        if tag:
            queryset = queryset.filter(tags__name__in=[tag])
        
        return queryset

```bash
        context.update(sidebar_context(self.request))
        
        # Добавляем навигацию

        context['previous_article'] = Article.objects.filter(

            is_approved=True

            comment.article = self.object
            comment.author = request.user
            comment.save()
            return redirect('blog:article_detail', slug=self.object.slug)
        
        context = self.get_context_data(**kwargs)
        context['comment_form'] = form
        return self.render_to_response(context)
```

## 🤖 **DeepSeek**

**Проблема найдена!** Ошибка: `Reverse for 'category_detail' with arguments '('',)' not found`. 

Шаблоны пытаются создать ссылку `{% url 'blog:category_detail' category.slug %}`, но у некоторых категорий `slug` пустой (`''`).

## 🔧 **БЫСТРОЕ ИСПРАВЛЕНИЕ:**

### **1. Проверить категории с пустыми slug:**

```bash

from uch.apps.blog.models import Category

print('Категории с пустыми или отсутствующими slug:')
for cat in Category.objects.all():

        print(f'  ID: {cat.id}, Name: \"{cat.name}\", Slug: \"{cat.slug}\"')
```

### **2. Временное решение - исправить шаблоны:**

```bash
# Найти все шаблоны, которые используют category_detail
find uch/apps/blog/templates -name "*.html" -type f -exec grep -l "category_detail" {} \;

# Временно заменить проблемные ссылки
for file in $(find uch/apps/blog/templates -name "*.html" -type f); do
```
